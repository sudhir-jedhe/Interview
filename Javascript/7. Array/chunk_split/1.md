JavaScript's standard library does not have an `Array.prototype.split` method (`split` belongs to `String.prototype`). In interviews and practical utilities, an "Array split polyfill" usually refers to one of two common requirements:

1. **Splitting an array by a delimiter / separator element or predicate** (mirroring `String.prototype.split`).
2. **Splitting an array into fixed-size chunks** (`chunk(size)`).

Here are both implementations, safe to attach to `Array.prototype` or use as standalone pure functions.

---

### 1. `Array.prototype.split` by Delimiter / Predicate (Mirrors `String.prototype.split`)

This splits an array wherever an element matches a given delimiter value or satisfies a predicate function. It also supports an optional `limit` parameter.

```javascript
if (!Array.prototype.split) {
  Object.defineProperty(Array.prototype, 'split', {
    value: function (delimiter, limit) {
      if (this == null) {
        throw new TypeError('Array.prototype.split called on null or undefined');
      }

      const list = Object(this);
      const len = list.length >>> 0;
      const maxChunks = limit === undefined ? Infinity : Math.max(0, limit >>> 0);

      if (maxChunks === 0) return [];

      const result = [];
      let currentChunk = [];

      const isMatch = typeof delimiter === 'function'
        ? delimiter
        : (item) => Object.is(item, delimiter);

      for (let i = 0; i < len; i++) {
        // Handle sparse arrays / holes
        const item = list[i];

        if (isMatch(item, i, list)) {
          result.push(currentChunk);
          currentChunk = [];

          if (result.length === maxChunks) {
            return result;
          }
        } else {
          currentChunk.push(item);
        }
      }

      // Push the trailing chunk if limit allows
      if (result.length < maxChunks) {
        result.push(currentChunk);
      }

      return result;
    },
    writable: true,
    configurable: true,
  });
}

```

#### Usage Examples

```javascript
// Example 1: Value delimiter
const tags = ['react', '|', 'vue', '|', 'angular'];
console.log(tags.split('|'));
// Output: [['react'], ['vue'], ['angular']]

// Example 2: Predicate delimiter
const numbers = [1, 2, 0, 4, 5, 0, 7];
console.log(numbers.split((num) => num === 0));
// Output: [[1, 2], [4, 5], [7]]

// Example 3: Limit argument
console.log(numbers.split((num) => num === 0, 2));
// Output: [[1, 2], [4, 5]]

```

### 2. `Array.prototype.chunk` (Split by Chunk Size)

If you meant dividing an array into sub-arrays of a fixed `chunkSize`:

```javascript
if (!Array.prototype.chunk) {
  Object.defineProperty(Array.prototype, 'chunk', {
    value: function (size = 1) {
      if (this == null) {
        throw new TypeError('Array.prototype.chunk called on null or undefined');
      }

      const chunkSize = Math.floor(size);
      if (chunkSize <= 0) {
        throw new RangeError('Chunk size must be a positive integer greater than 0');
      }

      const list = Object(this);
      const len = list.length >>> 0;
      const result = [];

      for (let i = 0; i < len; i += chunkSize) {
        result.push(Array.prototype.slice.call(list, i, i + chunkSize));
      }

      return result;
    },
    writable: true,
    configurable: true,
  });
}

```

#### Usage Examples

```javascript
const items = [1, 2, 3, 4, 5, 6, 7, 8];
console.log(items.chunk(3));
// Output: [[1, 2, 3], [4, 5, 6], [7, 8]]

```

### 3. Pure Function Variant (No Prototype Modification)

Modifying built-in prototypes in production codebases can lead to namespace pollution or collisions with future ECMAScript proposals. Here is the pure utility function:

```javascript
/**
 * Splits an array into sub-arrays by delimiter or predicate.
 * @template T
 * @param {T[]} array
 * @param {T | ((item: T, index: number, arr: T[]) => boolean)} delimiter
 * @param {number} [limit]
 * @returns {T[][]}
 */
export function splitArray(array, delimiter, limit = Infinity) {
  if (!Array.isArray(array)) {
    throw new TypeError('Expected an array');
  }

  const max = Math.max(0, limit >>> 0);
  if (max === 0) return [];

  const result = [];
  let current = [];
  const matches = typeof delimiter === 'function'
    ? delimiter
    : (val) => Object.is(val, delimiter);

  for (let i = 0; i < array.length; i++) {
    if (matches(array[i], i, array)) {
      result.push(current);
      current = [];
      if (result.length === max) return result;
    } else {
      current.push(array[i]);
    }
  }

  if (result.length < max) {
    result.push(current);
  }

  return result;
}

```

### Edge Cases Handled

* **Sparse arrays:** Safely steps through indices using `length >>> 0`.
* **`NaN` and `-0`:** Uses `Object.is` for value equality so `NaN` matches `NaN` and `+0` is distinguished from `-0`.
* **`Object.defineProperty`:** Marks the property non-enumerable (`enumerable: false` by default), preventing it from showing up in `for...in` loops over arrays.
