```javascript
/* ==========================================================================
   JAVASCRIPT TIMERS: COMPREHENSIVE CHEAT SHEET & REFERENCE
   ========================================================================== */

/* --------------------------------------------------------------------------
   1. CORE TIMER FUNCTIONS
   --------------------------------------------------------------------------
   • setTimeout(fn, delay, ...args): Schedules a function to run once after delay.
   • setInterval(fn, delay, ...args): Schedules a function to run repeatedly.
   • clearTimeout(id): Cancels a pending timeout.
   • clearInterval(id): Cancels an active interval.
   -------------------------------------------------------------------------- */

// Basic setTimeout
const timeoutId = setTimeout((greeting, name) => {
  console.log(`${greeting}, ${name}!`);
}, 1000, "Hello", "Alice"); // Arguments passed after delay

// Basic setInterval
let count = 0;
const intervalId = setInterval(() => {
  count++;
  console.log(`Tick ${count}`);
  if (count === 3) clearInterval(intervalId); // Stop interval
}, 1000);


/* --------------------------------------------------------------------------
   2. THE EVENT LOOP & TIMING GUARANTEES
   --------------------------------------------------------------------------
   • Timers specify a MINIMUM delay, not an exact time.
   • Macrotask vs Microtask Priority:
     - Microtasks (Promises, queueMicrotask) run BEFORE the next Macrotask (Timers).
     - Standard code execution on the Call Stack blocks timer execution.
   -------------------------------------------------------------------------- */

console.log("1. Sync");

// Macrotask (Timer)
setTimeout(() => console.log("4. Timer Macrotask"), 0);

// Microtask (Promise)
Promise.resolve().then(() => console.log("3. Promise Microtask"));

console.log("2. Sync End");

// Output Order:
// 1. Sync
// 2. Sync End
// 3. Promise Microtask
// 4. Timer Macrotask


/* --------------------------------------------------------------------------
   3. SETINTERVAL DRIFT VS. RECURSIVE SETTIMEOUT
   --------------------------------------------------------------------------
   • setInterval schedules execution without waiting for the prior task to finish.
     If task duration exceeds the delay, executions overlap ("interval drift").
   • Recursive setTimeout guarantees fixed delay AFTER task completion.
   -------------------------------------------------------------------------- */

// Flawed: Requests can overlap if fetchData takes > 1000ms
setInterval(async () => {
  await fetchData(); 
}, 1000);

// Recommended: Guarantees 1000ms pause BETWEEN complete operations
async function pollData() {
  await fetchData();
  setTimeout(pollData, 1000);
}
pollData();


/* --------------------------------------------------------------------------
   4. BROWSER THROTTLING & ANIMATIONS
   --------------------------------------------------------------------------
   • Inactive/Background Tabs: Browsers throttle setTimeout/setInterval 
     to >= 1000ms (or more) to save CPU/battery.
   • Visual Animations: Avoid timers for UI animations. Use requestAnimationFrame()
     for smooth 60fps rendering synced with display refresh rate.
   -------------------------------------------------------------------------- */

// Preferred approach for smooth, tab-aware animations
function animate(timestamp) {
  // Update animation state
  requestAnimationFrame(animate); // Automatically pauses in inactive tabs
}
requestAnimationFrame(animate);


/* --------------------------------------------------------------------------
   5. MODERN PROMISE-BASED DELAYS (ASYNC / AWAIT)
   --------------------------------------------------------------------------
   Convert callback-based timers into awaitable Promise delays.
   -------------------------------------------------------------------------- */

// Promise delay utility helper
const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

async function executeSequence() {
  console.log("Start process...");
  await sleep(2000); // Wait 2 seconds cleanly
  console.log("Process complete after 2s delay.");
}

executeSequence();


/* --------------------------------------------------------------------------
   6. COMMON TIMER PITFALLS
   -------------------------------------------------------------------------- */

// ❌ Trap 1: Invoking callback immediately
// setTimeout(myFunction(), 1000); // Runs immediately, passes return value!

// ✅ Fix: Pass function reference or arrow wrapper
// setTimeout(myFunction, 1000);
// setTimeout(() => myFunction(), 1000);

// ❌ Trap 2: String-based evaluation (eval security risk)
// setTimeout("console.log('Bad')", 1000);

// ❌ Trap 3: Unhandled multiple interval creation
// function startClock() { setInterval(showTime, 1000); } // Multi-clicks stack intervals!

// ✅ Fix: Guard against multiple interval instances
let activeInterval = null;
function startClockSafely() {
  if (!activeInterval) {
    activeInterval = setInterval(showTime, 1000);
  }
}
```
