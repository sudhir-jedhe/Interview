A **Config-Driven Form** (also known as a Schema-Driven Form) allows you to render complex, dynamic forms purely from a JSON specification. This decouples form UI/logic from the actual render code, making it incredibly easy to manage dynamic forms, survey builders, or backend-driven UI configurations.

Here is a clean, scalable implementation built from scratch using **React**, **TypeScript**, **React Hook Form**, and **Zod**.

---

## 1. Defining the Form Schema & Config Specification

First, define the JSON structure that specifies fields, types, options, dynamic visibility rules, and validation criteria.

```typescript
// types/formConfig.ts

export type FieldType = "text" | "number" | "select" | "checkbox" | "textarea";

export interface ValidationRule {
  required?: boolean;
  min?: number;
  max?: number;
  minLength?: number;
  pattern?: string;
  message?: string;
}

export interface ConditionalRule {
  field: string; // The field to watch (e.g., "accountType")
  equals: any; // The value that triggers visibility (e.g., "business")
}

export interface FieldConfig {
  name: string;
  label: string;
  type: FieldType;
  placeholder?: string;
  defaultValue?: any;
  options?: { label: string; value: string | number }[]; // For select inputs
  validation?: ValidationRule;
  dependsOn?: ConditionalRule; // For conditional rendering
}

export interface FormConfig {
  title: string;
  fields: FieldConfig[];
}
```

## 2. Dynamic Zod Schema Generator

Instead of hardcoding a static Zod schema, build a helper function that reads the `FormConfig` JSON array and generates a **dynamic Zod validation schema** on the fly.

```typescript
// utils/schemaBuilder.ts
import { z } from "zod";
import { FieldConfig } from "../types/formConfig";

export function buildDynamicZodSchema(fields: FieldConfig[]) {
  const shape: Record<string, z.ZodTypeAny> = {};

  fields.forEach((field) => {
    let validator: z.ZodTypeAny;

    // Base Type
    switch (field.type) {
      case "number":
        validator = z.coerce.number();
        break;
      case "checkbox":
        validator = z.boolean();
        break;
      default:
        validator = z.string();
    }

    // Apply Validation Rules
    if (field.validation) {
      const { required, minLength, min, max, pattern, message } =
        field.validation;

      if (
        field.type === "string" ||
        field.type === "text" ||
        field.type === "textarea"
      ) {
        if (minLength) {
          validator = (validator as z.ZodString).min(
            minLength,
            message || `Must be at least ${minLength} characters`,
          );
        }
        if (pattern) {
          validator = (validator as z.ZodString).regex(
            new RegExp(pattern),
            message || "Invalid format",
          );
        }
        if (required) {
          validator = (validator as z.ZodString).min(
            1,
            message || `${field.label} is required`,
          );
        } else {
          validator = validator.optional();
        }
      }

      if (field.type === "number") {
        if (min !== undefined)
          validator = (validator as z.ZodNumber).min(min, message);
        if (max !== undefined)
          validator = (validator as z.ZodNumber).max(max, message);
        if (!required) validator = validator.optional();
      }
    } else {
      validator = validator.optional();
    }

    shape[field.name] = validator;
  });

  return z.object(shape);
}
```

## 3. The Config-Driven Form Component

Now build a generic engine component (`ConfigForm`) that receives the JSON spec, evaluates conditional visibility dynamically via `useWatch`, and renders input controls accordingly.

```tsx
// components/ConfigForm.tsx
import React, { useMemo } from "react";
import { useForm, useWatch } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import { FormConfig, FieldConfig } from "../types/formConfig";
import { buildDynamicZodSchema } from "../utils/schemaBuilder";

interface ConfigFormProps {
  config: FormConfig;
  onSubmit: (data: any) => void;
}

// Sub-component to handle conditional field evaluation safely
function RenderField({
  field,
  register,
  errors,
  control,
}: {
  field: FieldConfig;
  register: any;
  errors: any;
  control: any;
}) {
  // Watch target field if conditional rule exists
  const watchedValue = field.dependsOn
    ? useWatch({ control, name: field.dependsOn.field })
    : null;

  // Evaluate conditional rule
  if (field.dependsOn && watchedValue !== field.dependsOn.equals) {
    return null; // Hide field if condition doesn't match
  }

  const errorMessage = errors[field.name]?.message as string | undefined;

  return (
    <div className="flex flex-col gap-1 mb-4">
      <label className="text-sm font-medium text-gray-700">{field.label}</label>

      {/* Render based on field type */}
      {field.type === "text" && (
        <input
          type="text"
          placeholder={field.placeholder}
          {...register(field.name)}
          className="border p-2 rounded focus:outline-blue-500"
        />
      )}

      {field.type === "number" && (
        <input
          type="number"
          placeholder={field.placeholder}
          {...register(field.name)}
          className="border p-2 rounded focus:outline-blue-500"
        />
      )}

      {field.type === "textarea" && (
        <textarea
          placeholder={field.placeholder}
          {...register(field.name)}
          className="border p-2 rounded focus:outline-blue-500"
        />
      )}

      {field.type === "select" && (
        <select
          {...register(field.name)}
          className="border p-2 rounded focus:outline-blue-500"
        >
          <option value="">Select option...</option>
          {field.options?.map((opt) => (
            <option key={opt.value} value={opt.value}>
              {opt.label}
            </option>
          ))}
        </select>
      )}

      {field.type === "checkbox" && (
        <div className="flex items-center gap-2">
          <input
            type="checkbox"
            id={field.name}
            {...register(field.name)}
            className="h-4 w-4"
          />
          <span className="text-sm">{field.placeholder}</span>
        </div>
      )}

      {errorMessage && (
        <span className="text-red-500 text-xs">{errorMessage}</span>
      )}
    </div>
  );
}

export function ConfigForm({ config, onSubmit }: ConfigFormProps) {
  // Build Zod Schema dynamically whenever config changes
  const dynamicSchema = useMemo(
    () => buildDynamicZodSchema(config.fields),
    [config],
  );

  // Extract default values from JSON config
  const defaultValues = useMemo(() => {
    const defaults: Record<string, any> = {};
    config.fields.forEach((f) => {
      defaults[f.name] = f.defaultValue ?? "";
    });
    return defaults;
  }, [config]);

  const {
    register,
    handleSubmit,
    control,
    formState: { errors, isSubmitting },
  } = useForm({
    resolver: zodResolver(dynamicSchema),
    defaultValues,
    mode: "onBlur",
  });

  return (
    <div className="max-w-lg mx-auto p-6 bg-white rounded-lg shadow-md">
      <h2 className="text-xl font-bold mb-4">{config.title}</h2>

      <form onSubmit={handleSubmit(onSubmit)}>
        {config.fields.map((field) => (
          <RenderField
            key={field.name}
            field={field}
            register={register}
            errors={errors}
            control={control}
          />
        ))}

        <button
          type="submit"
          disabled={isSubmitting}
          className="w-full mt-4 bg-blue-600 text-white py-2 rounded hover:bg-blue-700 transition"
        >
          {isSubmitting ? "Submitting..." : "Submit"}
        </button>
      </form>
    </div>
  );
}
```

## 4. Example Usage: Feeding JSON to the Component

Now you can build complex forms purely by defining or fetching JSON objects:

```tsx
// App.tsx
import React from "react";
import { ConfigForm } from "./components/ConfigForm";
import { FormConfig } from "./types/formConfig";

const sampleFormConfig: FormConfig = {
  title: "User Onboarding Form",
  fields: [
    {
      name: "accountType",
      label: "Account Type",
      type: "select",
      defaultValue: "personal",
      options: [
        { label: "Personal", value: "personal" },
        { label: "Business", value: "business" },
      ],
      validation: { required: true },
    },
    {
      name: "companyName",
      label: "Company Name",
      type: "text",
      placeholder: "Acme Corp",
      validation: {
        required: true,
        message: "Company name is mandatory for business accounts",
      },
      dependsOn: { field: "accountType", equals: "business" }, // Condition
    },
    {
      name: "fullName",
      label: "Full Name",
      type: "text",
      placeholder: "John Doe",
      validation: { required: true, minLength: 3 },
    },
    {
      name: "age",
      label: "Age",
      type: "number",
      placeholder: "18",
      validation: { min: 18, message: "Must be at least 18 years old" },
    },
    {
      name: "subscribe",
      label: "Newsletter",
      type: "checkbox",
      placeholder: "Subscribe to weekly updates",
    },
  ],
};

export default function App() {
  const handleFormSubmit = (data: any) => {
    console.log("Form Output:", data);
    alert(JSON.stringify(data, null, 2));
  };

  return <ConfigForm config={sampleFormConfig} onSubmit={handleFormSubmit} />;
}
```

### Key Advantages of This Architecture

1. **Zero UI Code Duplication:** To add, remove, or modify a field, you simply change a line in your JSON database or config payload—no JSX modifications required.
2. **Server-Driven UI:** You can store the JSON form configurations in a backend database (MongoDB/PostgreSQL) and send form specs over REST/GraphQL endpoints dynamically based on user roles or region settings.
3. **Optimized Render Cycles:** By isolating field evaluation into a `RenderField` sub-component using `useWatch`, changing field values triggers re-renders **only** for the conditional fields that depend on them.
