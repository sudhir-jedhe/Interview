## 1. Dynamic Form Renderer (Project #1)

Generates dynamic input elements and validations based on a JSON schema.

```tsx
import React, { useState } from "react";

interface FieldSchema {
  id: string;
  label: string;
  type: "text" | "number" | "select" | "checkbox";
  required?: boolean;
  options?: string[]; // For select type
}

interface FormSchema {
  title: string;
  fields: FieldSchema[];
}

const sampleSchema: FormSchema = {
  title: "User Onboarding Form",
  fields: [
    { id: "fullName", label: "Full Name", type: "text", required: true },
    { id: "age", label: "Age", type: "number", required: true },
    {
      id: "country",
      label: "Country",
      type: "select",
      options: ["USA", "India", "Canada"],
      required: true,
    },
    { id: "subscribe", label: "Subscribe to Newsletter", type: "checkbox" },
  ],
};

export const DynamicForm: React.FC = () => {
  const [formData, setFormData] = useState<Record<string, any>>({});
  const [errors, setErrors] = useState<Record<string, string>>({});

  const handleChange = (id: string, value: any) => {
    setFormData((prev) => ({ ...prev, [id]: value }));
    if (errors[id]) {
      setErrors((prev) => ({ ...prev, [id]: "" }));
    }
  };

  const validate = () => {
    const newErrors: Record<string, string> = {};
    sampleSchema.fields.forEach((field) => {
      const val = formData[field.id];
      if (
        field.required &&
        (val === undefined || val === "" || val === false)
      ) {
        newErrors[field.id] = `${field.label} is required`;
      }
    });
    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (validate()) {
      alert(
        `Form Submitted Successfully!\n${JSON.stringify(formData, null, 2)}`,
      );
    }
  };

  return (
    <div
      style={{
        maxWidth: "400px",
        margin: "20px auto",
        fontFamily: "sans-serif",
      }}
    >
      <h2>{sampleSchema.title}</h2>
      <form onSubmit={handleSubmit}>
        {sampleSchema.fields.map((field) => (
          <div key={field.id} style={{ marginBottom: "15px" }}>
            <label
              style={{
                display: "block",
                marginBottom: "5px",
                fontWeight: "bold",
              }}
            >
              {field.label} {field.required && "*"}
            </label>

            {field.type === "text" && (
              <input
                type="text"
                value={formData[field.id] || ""}
                onChange={(e) => handleChange(field.id, e.target.value)}
                style={{
                  width: "100%",
                  padding: "8px",
                  boxSizing: "border-box",
                }}
              />
            )}

            {field.type === "number" && (
              <input
                type="number"
                value={formData[field.id] || ""}
                onChange={(e) => handleChange(field.id, e.target.value)}
                style={{
                  width: "100%",
                  padding: "8px",
                  boxSizing: "border-box",
                }}
              />
            )}

            {field.type === "select" && (
              <select
                value={formData[field.id] || ""}
                onChange={(e) => handleChange(field.id, e.target.value)}
                style={{
                  width: "100%",
                  padding: "8px",
                  boxSizing: "border-box",
                }}
              >
                <option value="">Select an option</option>
                {field.options?.map((opt) => (
                  <option key={opt} value={opt}>
                    {opt}
                  </option>
                ))}
              </select>
            )}

            {field.type === "checkbox" && (
              <input
                type="checkbox"
                checked={!!formData[field.id]}
                onChange={(e) => handleChange(field.id, e.target.checked)}
              />
            )}

            {errors[field.id] && (
              <span
                style={{
                  color: "red",
                  fontSize: "12px",
                  display: "block",
                  marginTop: "4px",
                }}
              >
                {errors[field.id]}
              </span>
            )}
          </div>
        ))}

        <button
          type="submit"
          style={{
            padding: "10px 20px",
            background: "#2563eb",
            color: "#fff",
            border: "none",
            borderRadius: "4px",
            cursor: "pointer",
          }}
        >
          Submit
        </button>
      </form>
    </div>
  );
};
```

## 2. In-Browser JavaScript Code Judge (Project #2)

Sandboxes JS code execution in the browser using dynamic evaluation and captures custom console output.

```tsx
import React, { useState } from "react";

export const JSOnlineJudge: React.FC = () => {
  const [code, setCode] = useState<string>(
    `// Problem: Return sum of array\nfunction solve(arr) {\n  return arr.reduce((a, b) => a + b, 0);\n}`,
  );
  const [output, setOutput] = useState<string[]>([]);
  const [testResult, setTestResult] = useState<string | null>(null);

  const runCode = () => {
    const logs: string[] = [];

    // Override console.log to intercept outputs
    const originalLog = console.log;
    console.log = (...args) => {
      logs.push(
        args
          .map((a) => (typeof a === "object" ? JSON.stringify(a) : a))
          .join(" "),
      );
      originalLog(...args);
    };

    try {
      // Safely evaluate function
      const userFunc = new Function(`${code}; return solve;`)();

      // Test Case
      const testInput = [1, 2, 3, 4, 5];
      const expectedOutput = 15;
      const result = userFunc(testInput);

      if (result === expectedOutput) {
        setTestResult(`✅ Test Passed! Input: [1,2,3,4,5] | Result: ${result}`);
      } else {
        setTestResult(
          `❌ Test Failed. Expected: ${expectedOutput}, Received: ${result}`,
        );
      }
    } catch (err: any) {
      setTestResult(`💥 Execution Error: ${err.message}`);
    } finally {
      console.log = originalLog; // Restore console
      setOutput(logs);
    }
  };

  return (
    <div
      style={{
        maxWidth: "700px",
        margin: "20px auto",
        fontFamily: "monospace",
      }}
    >
      <h2>In-Browser JS Judge</h2>
      <textarea
        rows={10}
        value={code}
        onChange={(e) => setCode(e.target.value)}
        style={{
          width: "100%",
          padding: "10px",
          background: "#1e293b",
          color: "#f8fafc",
          borderRadius: "6px",
        }}
      />
      <button
        onClick={runCode}
        style={{
          marginTop: "10px",
          padding: "10px 20px",
          background: "#16a34a",
          color: "#fff",
          border: "none",
          cursor: "pointer",
          borderRadius: "4px",
        }}
      >
        Run Test Cases
      </button>

      {testResult && (
        <div
          style={{
            marginTop: "15px",
            padding: "12px",
            background: "#f1f5f9",
            borderRadius: "4px",
            fontWeight: "bold",
          }}
        >
          {testResult}
        </div>
      )}

      {output.length > 0 && (
        <div
          style={{
            marginTop: "15px",
            background: "#0f172a",
            color: "#38bdf8",
            padding: "10px",
            borderRadius: "4px",
          }}
        >
          <strong>Console Output:</strong>
          {output.map((line, i) => (
            <div key={i}>{line}</div>
          ))}
        </div>
      )}
    </div>
  );
};
```

## 4. Chrome's Dino Canvas Game (Project #28)

HTML5 Canvas loop using `requestAnimationFrame` with collision detection and jumping logic.

```tsx
import React, { useRef, useEffect, useState } from "react";

export const DinoGame: React.FC = () => {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const [gameOver, setGameOver] = useState(false);
  const [score, setScore] = useState(0);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    let animationId: number;
    let scoreCount = 0;

    // Game state
    const dino = {
      x: 50,
      y: 150,
      width: 30,
      height: 30,
      dy: 0,
      gravity: 0.6,
      jumpPower: -10,
      grounded: true,
    };
    const obstacle = { x: 600, y: 150, width: 20, height: 30, speed: 5 };

    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.code === "Space" && dino.grounded) {
        dino.dy = dino.jumpPower;
        dino.grounded = false;
      }
    };

    window.addEventListener("keydown", handleKeyDown);

    const gameLoop = () => {
      // Clear canvas
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      // Update Dino position
      dino.dy += dino.gravity;
      dino.y += dino.dy;

      if (dino.y >= 150) {
        dino.y = 150;
        dino.dy = 0;
        dino.grounded = true;
      }

      // Move Obstacle
      obstacle.x -= obstacle.speed;
      if (obstacle.x < -obstacle.width) {
        obstacle.x = canvas.width;
        scoreCount += 10;
        setScore(scoreCount);
      }

      // Collision Detection
      if (
        dino.x < obstacle.x + obstacle.width &&
        dino.x + dino.width > obstacle.x &&
        dino.y < obstacle.y + obstacle.height &&
        dino.y + dino.height > obstacle.y
      ) {
        setGameOver(true);
        return; // Stop animation loop
      }

      // Draw Ground
      ctx.beginPath();
      ctx.moveTo(0, 180);
      ctx.lineTo(600, 180);
      ctx.stroke();

      // Draw Dino (Green Square)
      ctx.fillStyle = "#16a34a";
      ctx.fillRect(dino.x, dino.y, dino.width, dino.height);

      // Draw Obstacle (Red Square)
      ctx.fillStyle = "#dc2626";
      ctx.fillRect(obstacle.x, obstacle.y, obstacle.width, obstacle.height);

      animationId = requestAnimationFrame(gameLoop);
    };

    animationId = requestAnimationFrame(gameLoop);

    return () => {
      cancelAnimationFrame(animationId);
      window.removeEventListener("keydown", handleKeyDown);
    };
  }, [gameOver]);

  return (
    <div
      style={{
        textAlign: "center",
        marginTop: "20px",
        fontFamily: "sans-serif",
      }}
    >
      <h2>Chrome Dino Game (Press Space to Jump)</h2>
      <p style={{ fontSize: "18px", fontWeight: "bold" }}>Score: {score}</p>
      <canvas
        ref={canvasRef}
        width={600}
        height={200}
        style={{ border: "2px solid #000", background: "#f8fafc" }}
      />
      {gameOver && (
        <div>
          <h3 style={{ color: "red" }}>Game Over!</h3>
          <button
            onClick={() => setGameOver(false)}
            style={{ padding: "8px 16px", cursor: "pointer" }}
          >
            Restart
```

## 3. Tic-Tac-Toe with Unbeatable Minimax Bot (Project #26)

Features a recursive **Minimax AI** decision tree that guarantees the bot never loses.

```tsx
import React, { useState } from "react";

type BoardState = (string | null)[];

export const TicTacToeBot: React.FC = () => {
  const [board, setBoard] = useState<BoardState>(Array(9).fill(null));
  const [winner, setWinner] = useState<string | null>(null);

  const checkWinner = (b: BoardState): string | null => {
    const lines = [
      [0, 1, 2],
      [3, 4, 5],
      [6, 7, 8],
      [0, 3, 6],
      [1, 4, 7],
      [2, 5, 8],
      [0, 4, 8],
      [2, 4, 6],
    ];
    for (const [a, bIdx, c] of lines) {
      if (b[a] && b[a] === b[bIdx] && b[a] === b[c]) return b[a];
    }
    return b.includes(null) ? null : "Tie";
  };

  const minimax = (
    b: BoardState,
    depth: number,
    isMax: boolean,
  ): { score: number; move?: number } => {
    const res = checkWinner(b);
    if (res === "O") return { score: 10 - depth };
    if (res === "X") return { score: depth - 10 };
    if (res === "Tie") return { score: 0 };

    const avail = b
      .map((v, i) => (v === null ? i : null))
      .filter((v): v is number => v !== null);

    if (isMax) {
      let bestScore = -Infinity;
      let bestMove = avail[0];
      for (const idx of avail) {
        b[idx] = "O";
        const { score } = minimax(b, depth + 1, false);
        b[idx] = null;
        if (score > bestScore) {
          bestScore = score;
          bestMove = idx;
        }
      }
      return { score: bestScore, move: bestMove };
    } else {
      let bestScore = Infinity;
      let bestMove = avail[0];
      for (const idx of avail) {
        b[idx] = "X";
        const { score } = minimax(b, depth + 1, true);
        b[idx] = null;
        if (score < bestScore) {
          bestScore = score;
          bestMove = idx;
        }
      }
      return { score: bestScore, move: bestMove };
    }
  };

  const handleClick = (idx: number) => {
    if (board[idx] || winner) return;

    const newBoard = [...board];
    newBoard[idx] = "X"; // Player Move

    let gameRes = checkWinner(newBoard);
    if (gameRes) {
      setBoard(newBoard);
      setWinner(gameRes);
      return;
    }

    // Bot Move (O)
    const { move } = minimax(newBoard, 0, true);
    if (move !== undefined) newBoard[move] = "O";

    gameRes = checkWinner(newBoard);
    setBoard(newBoard);
    if (gameRes) setWinner(gameRes);
  };

  return (
    <div
      style={{
        textAlign: "center",
        fontFamily: "sans-serif",
        marginTop: "20px",
      }}
    >
      <h2>Tic-Tac-Toe vs Unbeatable Bot</h2>
      <div
        style={{
          display: "grid",
          gridTemplateColumns: "repeat(3, 80px)",
          gap: "6px",
          justifyContent: "center",
        }}
      >
        {board.map((cell, idx) => (
          <button
            key={idx}
            onClick={() => handleClick(idx)}
            style={{
              width: "80px",
              height: "80px",
              fontSize: "28px",
              fontWeight: "bold",
              cursor: "pointer",
            }}
          >
            {cell}
          </button>
        ))}
      </div>
      {winner && (
        <h3 style={{ marginTop: "20px" }}>
          {winner === "Tie" ? "It is a Tie!" : `Winner: ${winner}`}
        </h3>
      )}
    </div>
  );
};
```

## 5. 2048 Game Engine (Project #27)

Grid slide-and-merge mechanics for 2048.

```tsx
import React, { useState, useEffect, useCallback } from "react";

type Grid = number[][];

export const Game2048: React.FC = () => {
  const createEmptyGrid = (): Grid =>
    Array(4)
      .fill(0)
      .map(() => Array(4).fill(0));

  const addNumber = (grid: Grid): Grid => {
    const newGrid = grid.map((r) => [...r]);
    const emptyCells: [number, number][] = [];
    for (let r = 0; r < 4; r++) {
      for (let c = 0; c < 4; c++) {
        if (newGrid[r][c] === 0) emptyCells.push([r, c]);
      }
    }
    if (emptyCells.length === 0) return newGrid;
    const [r, c] = emptyCells[Math.floor(Math.random() * emptyCells.length)];
    newGrid[r][c] = Math.random() < 0.9 ? 2 : 4;
    return newGrid;
  };

  const [grid, setGrid] = useState<Grid>(() =>
    addNumber(addNumber(createEmptyGrid())),
  );

  const slideLeft = (row: number[]): number[] => {
    let arr = row.filter((val) => val !== 0);
    for (let i = 0; i < arr.length - 1; i++) {
      if (arr[i] === arr[i + 1]) {
        arr[i] *= 2;
        arr[i + 1] = 0;
      }
    }
    arr = arr.filter((val) => val !== 0);
    while (arr.length < 4) arr.push(0);
    return arr;
  };

  const moveLeft = useCallback(() => {
    let moved = false;
    const newGrid = grid.map((row) => {
      const newRow = slideLeft(row);
      if (newRow.join(",") !== row.join(",")) moved = true;
      return newRow;
    });
    if (moved) setGrid(addNumber(newGrid));
  }, [grid]);

  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === "ArrowLeft") moveLeft();
    };
    window.addEventListener("keydown", handleKeyDown);
    return () => window.removeEventListener("keydown", handleKeyDown);
  }, [moveLeft]);

  return (
    <div
      style={{
        textAlign: "center",
        fontFamily: "sans-serif",
        marginTop: "20px",
```
