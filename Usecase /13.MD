Here is a complete, working hands-on guide featuring **Node.js (Express)** backend implementations and **React (Fetch / Axios)** client consumption patterns for all 18 common HTTP status codes.

---

## 1. Success Codes (2xx)

### `200 OK` — Request succeeded

- **Backend (Node.js):**

```javascript
app.get("/api/users/profile", (req, res) => {
  res.status(200).json({ id: 1, name: "Alice", role: "admin" });
});
```

- **Frontend (React):**

```jsx
useEffect(() => {
  fetch("/api/users/profile")
    .then((res) => {
      if (res.status === 200) return res.json();
    })
    .then((data) => setUser(data));
}, []);
```

### `201 Created` — Resource created successfully

- **Backend (Node.js):**

```javascript
app.post("/api/products", (req, res) => {
  const newProduct = { id: Date.now(), ...req.body };
  res.status(201).json(newProduct);
});
```

- **Frontend (React):**

```jsx
const createProduct = async (productData) => {
  const res = await fetch("/api/products", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(productData),
  });
  if (res.status === 201) {
    const created = await res.json();
    setProducts((prev) => [...prev, created]);
  }
};
```

### `202 Accepted` — Request accepted for asynchronous processing

- **Backend (Node.js):**

```javascript
app.post("/api/export-pdf", (req, res) => {
  // Queue background task (e.g., BullMQ / AWS SQS)
  queuePdfGeneration(req.body.userId);
  res.status(202).json({ message: "Export queued", jobId: "job_12345" });
});
```

- **Frontend (React):**

```jsx
const triggerExport = async () => {
  const res = await fetch("/api/export-pdf", { method: "POST" });
  if (res.status === 202) {
    const data = await res.json();
    setStatus(`Processing... Job ID: ${data.jobId}`);
  }
};
```

### `204 No Content` — Success, but no content body returned

- **Backend (Node.js):**

```javascript
app.delete("/api/items/:id", (req, res) => {
  deleteItemFromDb(req.params.id);
  res.status(204).send(); // Note: .send() without body
});
```

- **Frontend (React):**

```jsx
const deleteItem = async (id) => {
  const res = await fetch(`/api/items/${id}`, { method: "DELETE" });
  if (res.status === 204) {
    setItems((prev) => prev.filter((item) => item.id !== id));
  }
};
```

## 2. Redirection Codes (3xx)

### `301 Moved Permanently` — Resource URL changed permanently

- **Backend (Node.js):**

```javascript
app.get("/old-api/v1/users", (req, res) => {
  res.redirect(301, "/api/v2/users");
});
```

- **Frontend (React):**
  > _Note: Browsers automatically follow 301 redirects transparently. The fetch response URL updates to the target._

```jsx
const fetchUsers = async () => {
  const res = await fetch("/old-api/v1/users");
  console.log(res.redirected); // true
  console.log(res.url); // "https://api.mysite.com/api/v2/users"
  const data = await res.json();
};
```

### `302 Found` — Temporarily moved

- **Backend (Node.js):**

```javascript
app.get("/login-oauth", (req, res) => {
  res.redirect(302, "https://auth.provider.com/login");
});
```

- **Frontend (React):**

```jsx
const handleOAuthLogin = () => {
  // Direct window location navigation for temporary auth redirects
  window.location.href = "/login-oauth";
};
```

### `304 Not Modified` — Use client's cached copy

- **Backend (Node.js):**

```javascript
app.get("/api/static-config", (req, res) => {
  const clientEtag = req.headers["if-none-match"];
  const currentEtag = "v1.0.2";

  if (clientEtag === currentEtag) {
    return res.status(304).send(); // Tell browser to use local cache
  }

  res.setHeader("ETag", currentEtag);
  res.status(200).json({ theme: "dark", version: "1.0.2" });
});
```

- **Frontend (React):**

```jsx
// Browsers automatically handle 304 and supply cached body transparently
useEffect(() => {
  fetch("/api/static-config")
    .then((res) => res.json())
    .then((config) => setConfig(config));
}, []);
```

## 3. Client Error Codes (4xx)

### `400 Bad Request` — Malformed request body or missing fields

- **Backend (Node.js):**

```javascript
app.post("/api/register", (req, res) => {
  const { email, password } = req.body;
  if (!email || !password) {
    return res.status(400).json({ error: "Email and password are required." });
  }
  // ... proceed with registration
});
```

- **Frontend (React):**

```jsx
const handleRegister = async (formData) => {
  const res = await fetch("/api/register", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(formData),
  });

  if (res.status === 400) {
    const { error } = await res.json();
    setValidationError(error); // Render field error in UI
  }
};
```

### `401 Unauthorized` — Authentication missing or invalid token

- **Backend (Node.js):**

```javascript
app.get("/api/dashboard", (req, res) => {
  const token = req.headers.authorization;
  if (!token || token !== "Bearer valid_jwt_token") {
    return res
      .status(401)
      .json({ error: "Authentication required. Token missing or invalid." });
  }
  res.json({ stats: [1, 2, 3] });
});
```

- **Frontend (React):**

```jsx
const fetchDashboard = async () => {
  const res = await fetch("/api/dashboard", {
    headers: { Authorization: `Bearer ${localStorage.getItem("token")}` },
  });

  if (res.status === 401) {
    // Clear token and redirect user to Login page
    localStorage.removeItem("token");
    navigate("/login");
  }
};
```

### `403 Forbidden` — Authenticated, but lacks permission

- **Backend (Node.js):**

```javascript
app.delete("/api/admin/users/:id", (req, res) => {
  if (req.user.role !== "SUPER_ADMIN") {
    return res
      .status(403)
      .json({ error: "Access denied. Administrator privileges required." });
  }
  // Delete logic...
});
```

- **Frontend (React):**

```jsx
const deleteUser = async (userId) => {
  const res = await fetch(`/api/admin/users/${userId}`, { method: "DELETE" });
  if (res.status === 403) {
    alert("You do not have permission to delete users.");
  }
};
```

### `404 Not Found` — Resource does not exist

- **Backend (Node.js):**

```javascript
app.get("/api/orders/:id", (req, res) => {
  const order = findOrderInDb(req.params.id);
  if (!order) {
    return res.status(404).json({ error: "Requested order does not exist." });
  }
  res.json(order);
});
```

- **Frontend (React):**

```jsx
const loadOrder = async (orderId) => {
  const res = await fetch(`/api/orders/${orderId}`);
  if (res.status === 404) {
    setNotFound(true); // Display "Order Not Found" component
  }
};
```

### `405 Method Not Allowed` — HTTP method supported, but not on this route

- **Backend (Node.js):**

```javascript
app
  .route("/api/contact")
  .post((req, res) => res.json({ success: true }))
  .all((req, res) => {
    res.setHeader("Allow", "POST");
    res
      .status(405)
      .json({ error: `Method ${req.method} not allowed on this route.` });
  });
```

- **Frontend (React):**

```jsx
const updateContact = async () => {
  // Incorrectly using PUT on a POST-only endpoint
  const res = await fetch("/api/contact", { method: "PUT" });
  if (res.status === 405) {
    console.error("API endpoint does not support PUT operations.");
  }
};
```

### `408 Request Timeout` — Server timed out waiting for request

- **Backend (Node.js):**

```javascript
app.post("/api/large-upload", (req, res) => {
  req.setTimeout(5000, () => {
    res.status(408).json({ error: "Request body transfer timed out." });
  });
});
```

- **Frontend (React):**

```jsx
const uploadData = async (payload) => {
  const res = await fetch("/api/large-upload", {
    method: "POST",
    body: payload,
  });
  if (res.status === 408) {
    alert("Network connection too slow. Please retry upload.");
  }
};
```

### `502 Bad Gateway` — Proxy / Gateway received an invalid upstream response

- **Backend (Node.js / Express Gateway / NGINX setup):**

    // Calling internal upstream microservice
    const microserviceRes = await fetch("http://internal-service:8080");

      .status(502)
      .json({

  }
});

- **Frontend (React):**

- **Backend (Node.js):**

app.use((req, res, next) => {

### `504 Gateway Timeout` — Gateway timed out waiting for upstream server

- **Backend (Node.js API Gateway):**

```javascript
app.get("/api/slow-report", async (req, res) => {
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), 3000); // 3s gateway timeout

  try {
    const reportRes = await fetch("http://slow-database-service/report", {
      signal: controller.signal,
    });
    const data = await reportRes.json();
    res.json(data);
  } catch (err) {
    if (err.name === "AbortError") {
      res
        .status(504)
        .json({
          error: "Gateway Timeout: Upstream database took too long to respond.",
        });
    }
  } finally {
    clearTimeout(timeout);
  }
});
```

- **Frontend (React):**

```jsx
const generateReport = async () => {
  const res = await fetch("/api/slow-report");
  if (res.status === 504) {
    alert(
      "The report request timed out on the server. Try narrowing your query date range.",
    );
  }
};
```

### 1. Node.js (Express) Centralized Error Handler

Rather than wrapping every controller in `try/catch` manually, use an async error wrapper and a unified error-handling middleware.

```javascript
// utils/AppError.js - Custom Error Class
class AppError extends Error {
  constructor(message, statusCode) {
    super(message);
    this.statusCode = statusCode;
    this.status = `${statusCode}`.startsWith("4") ? "fail" : "error";
    this.isOperational = true; // Distinguishes operational errors from bugs
    Error.captureStackTrace(this, this.constructor);
  }
}

// utils/catchAsync.js - Higher-order function to remove boilerplate try/catch
const catchAsync = (fn) => (req, res, next) => {
  fn(req, res, next).catch(next);
};

// server.js / app.js
import express from "express";

const app = express();
app.use(express.json());

// Sample Routes throwing explicit HTTP errors
app.get(
  "/api/users/:id",
  catchAsync(async (req, res, next) => {
    const user = await findUserInDb(req.params.id);
    if (!user) {
      return next(new AppError("User not found", 404)); // Triggers 404
    }
    res.status(200).json({ status: "success", data: user });
  }),
);

app.post(
  "/api/admin/action",
  catchAsync(async (req, res, next) => {
    if (req.user?.role !== "ADMIN") {
      return next(new AppError("Forbidden: Admin access required", 403)); // Triggers 403
    }
    res.status(200).json({ message: "Success" });
  }),
);

// Global Error Handling Middleware (Catches ALL errors)
app.use((err, req, res, next) => {
  err.statusCode = err.statusCode || 500;
  err.status = err.status || "error";

  // Log unexpected internal bugs (500s) for monitoring
  if (err.statusCode === 500) {
    console.error("CRITICAL UNHANDLED ERROR 💥:", err);
  }

  res.status(err.statusCode).json({
    status: err.status,
    message: err.message || "Internal Server Error",
    ...(process.env.NODE_ENV === "development" && { stack: err.stack }),
  });
});
```

### 2. React Client Unified Axios Interceptor

Using an Axios instance with interceptors allows you to handle global statuses (like `401 Unauthorized` or `503 Service Unavailable`) automatically in one place, while letting UI components handle localized status codes (like `400` or `404`).

```javascript
// src/api/axiosClient.js
import axios from "axios";

const api = axios.create({
  baseURL: "/api",
  timeout: 10000, // 10s default timeout
});

// Request Interceptor: Attach JWT Token automatically
api.interceptors.request.use((config) => {
  const token = localStorage.getItem("token");
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// Response Interceptor: Centralized Error Handling across ALL HTTP Statuses
api.interceptors.response.use(
  (response) => {
    // Passes through successful 2xx responses
    return response;
  },
  (error) => {
    // Handle Network / Timeout Errors (No Response Received)
    if (!error.response) {
      if (error.code === "ECONNABORTED") {
        console.error("Request timed out (408/504 style client abort)");
      } else {
        console.error("Network failure - Server is completely unreachable.");
      }
      return Promise.reject(error);
    }

    const { status, data } = error.response;

    // Centralized Status Code Switch
    switch (status) {
      case 401: // Unauthorized
        console.warn("Session expired. Redirecting to login...");
        localStorage.removeItem("token");
        window.location.href = "/login";
        break;

      case 403: // Forbidden
        console.warn("Access denied:", data.message);
        break;

      case 500: // Server Crash
        console.error("Server Internal Error:", data.message);
        break;

      case 503: // Service Unavailable
        alert("Service is currently down for scheduled maintenance.");
        break;

      default:
        // Other statuses (400, 404, etc.) are passed to the component
        break;
    }

    return Promise.reject(error);
  },
);

export default api;
```

### 3. Consuming in a React Component with UI Error States

```jsx
// src/components/UserProfile.jsx
import React, { useState, useEffect } from "react";
import api from "../api/axiosClient";

export default function UserProfile({ userId }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [errorMessage, setErrorMessage] = useState("");

  useEffect(() => {
    const fetchProfile = async () => {
      setLoading(true);
      setErrorMessage("");

      try {
        const response = await api.get(`/users/${userId}`);
        setUser(response.data.data); // Handles 200 OK
      } catch (err) {
        if (err.response) {
          const status = err.response.status;

          // Component-specific error handling
          if (status === 404) {
            setErrorMessage("User profile not found.");
          } else if (status === 400) {
            setErrorMessage("Invalid User ID supplied.");
          } else {
            setErrorMessage(err.response.data.message || "An error occurred.");
          }
        } else {
          setErrorMessage("Network error. Please check your connection.");
        }
      } finally {
        setLoading(false);
      }
    };

    fetchProfile();
  }, [userId]);

  if (loading) return <div>Loading user profile...</div>;
  if (errorMessage) return <div className="error-banner">{errorMessage}</div>;

  return (
    <div>
      <h1>{user.name}</h1>
      <p>Role: {user.role}</p>
    </div>
  );
}
```
