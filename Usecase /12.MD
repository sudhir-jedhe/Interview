### 1. Product of Array Except Self

```javascript
function productExceptSelf(nums) {
  const n = nums.length;
  const res = new Array(n).fill(1);

  // Calculate prefix products
  let prefix = 1;
  for (let i = 0; i < n; i++) {
    res[i] = prefix;
    prefix *= nums[i];
  }

  // Calculate suffix products and combine
  let suffix = 1;
  for (let i = n - 1; i >= 0; i--) {
    res[i] *= suffix;
    suffix *= nums[i];
  }

  return res;
}

// Example: productExceptSelf([1, 2, 3, 4]) -> [24, 12, 8, 6]
```

### 2. Word / String Pattern Matching (Isomorphic Strings)

```javascript
function isIsomorphic(s, t) {
  if (s.length !== t.length) return false;

  const s2t = new Map();
  const t2s = new Map();

  for (let i = 0; i < s.length; i++) {
    const charS = s[i];
    const charT = t[i];

    if (
      (s2t.has(charS) && s2t.get(charS) !== charT) ||
      (t2s.has(charT) && t2s.get(charT) !== charS)
    ) {
      return false;
    }

    s2t.set(charS, charT);
    t2s.set(charT, charS);
  }

  return true;
}

// Example: isIsomorphic("egg", "add") -> true
```

### 3. Separate Even and Odd Numbers

```javascript
function separateEvenOdd(nums) {
  let left = 0;
  let right = nums.length - 1;

  while (left < right) {
    if (nums[left] % 2 === 0) {
      left++;
    } else if (nums[right] % 2 !== 0) {
      right--;
    } else {
      [nums[left], nums[right]] = [nums[right], nums[left]];
      left++;
      right--;
    }
  }

  return nums;
}

// Example: separateEvenOdd([3, 1, 2, 4, 7, 6]) -> [6, 4, 2, 7, 1, 3]
```

### 4. Valid Balanced Parentheses / Brackets

```javascript
function isValidBrackets(s) {
  const stack = [];
  const map = { ")": "(", "}": "{", "]": "[" };

  for (const char of s) {
    if (char in map) {
      const topElement = stack.length > 0 ? stack.pop() : "#";
      if (map[char] !== topElement) return false;
    } else {
      stack.push(char);
    }
  }

  return stack.length === 0;
}

// Example: isValidBrackets("{[()]}") -> true
```

### 5. Generate All Subsets (Power Set)

```javascript
function subsets(nums) {
  const res = [];

  function backtrack(start, path) {
    res.push([...path]);
    for (let i = start; i < nums.length; i++) {
      path.push(nums[i]);
      backtrack(i + 1, path);
      path.pop();
    }
  }

  backtrack(0, []);
  return res;
}

// Example: subsets([1, 2]) -> [[], [1], [1, 2], [2]]
```

### 7. Subarray Sum Equals K

```javascript
function subarraySum(nums, k) {
  let count = 0;
  let currentSum = 0;
  const prefixSums = new Map();
  prefixSums.set(0, 1); // Base case: sum of 0 occurs once

  for (const num of nums) {
    currentSum += num;
    if (prefixSums.has(currentSum - k)) {
      count += prefixSums.get(currentSum - k);
    }
    prefixSums.set(currentSum, (prefixSums.get(currentSum) || 0) + 1);
  }

  return count;
}

// Example: subarraySum([1, 1, 1], 2) -> 2
```

### 8. Find the Longest Word in a String

```javascript
function longestWord(s) {
  const words = s.match(/\b\w+\b/g) || [];
  let longest = "";

  for (const word of words) {
    if (word.length > longest.length) {
      longest = word;
    }
  }

  return longest;
}

// Example: longestWord("The quick brown fox jumps over the lazy dog!") -> "jumps"
```

### 9. 3Sum (Unique Triplets summing to zero)

```javascript
function threeSum(nums) {
  nums.sort((a, b) => a - b);
  const res = [];

  for (let i = 0; i < nums.length - 2; i++) {
    if (i > 0 && nums[i] === nums[i - 1]) continue;

    let left = i + 1;
    let right = nums.length - 1;

    while (left < right) {
      const sum = nums[i] + nums[left] + nums[right];

      if (sum < 0) {
        left++;
      } else if (sum > 0) {
        right--;
      } else {
        res.push([nums[i], nums[left], nums[right]]);
        while (left < right && nums[left] === nums[left + 1]) left++;
        while (left < right && nums[right] === nums[right - 1]) right--;
        left++;
        right--;
      }
    }
  }

  return res;
}

// Example: threeSum([-1, 0, 1, 2, -1, -4]) -> [[-1, -1, 2], [-1, 0, 1]]
```

### 10. String Compression

```javascript
function compressString(s) {
  if (!s) return "";

  let res = "";
  let count = 1;

  for (let i = 1; i < s.length; i++) {
    if (s[i] === s[i - 1]) {
      count++;
    } else {
      res += s[i - 1] + count;
      count = 1;
    }
  }

  res += s[s.length - 1] + count;
  return res;
}

// Example: compressString("aabbb") -> "a2b3"
```

### 11. Reverse an Array In-Place

```javascript
function reverseArray(arr) {
  let left = 0;
  let right = arr.length - 1;

  while (left < right) {
    [arr[left], arr[right]] = [arr[right], arr[left]];
    left++;
    right--;
  }

  return arr;
}

// Example: reverseArray([1, 2, 3, 4, 5]) -> [5, 4, 3, 2, 1]
```

### 12. Remove Characters of Second String from First

```javascript
function removeChars(str1, str2) {
  const charsToRemove = new Set(str2);
  let res = "";

  for (const char of str1) {
    if (!charsToRemove.has(char)) {
      res += char;
    }
  }

  return res;
}

// Example: removeChars("hello world", "elo") -> "h wrd"
```

### 13. Longest Increasing Subsequence (LIS)

```javascript
function lengthOfLIS(nums) {
  if (nums.length === 0) return 0;

  const tails = [];

  for (const num of nums) {
    let left = 0;
    let right = tails.length;

    // Binary search for placement index
    while (left < right) {
      const mid = Math.floor((left + right) / 2);
      if (tails[mid] < num) {
        left = mid + 1;
      } else {
        right = mid;
      }
    }

    tails[left] = num;
  }

  return tails.length;
}

// Example: lengthOfLIS([10, 9, 2, 5, 3, 7, 101, 18]) -> 4 (Subsequence: [2, 3, 7, 101])
```

### 14. Determine if One String is a Rotation of Another

```javascript
function isRotation(s1, s2) {
  if (s1.length !== s2.length || s1.length === 0) return false;
  return (s1 + s1).includes(s2);
}

// Example: isRotation("waterbottle", "erbottlewat") -> true
```

### 15. Count Pairs with Difference Equal to K

```javascript
function countPairsWithDiffK(nums, k) {
  if (k < 0) return 0;

  const counts = new Map();
  for (const num of nums) {
    counts.set(num, (counts.get(num) || 0) + 1);
  }

  let count = 0;

  for (const [num, numCount] of counts) {
    if (k === 0) {
      if (numCount > 1) count++;
    } else {
      if (counts.has(num + k)) count++;
    }
  }

  return count;
}

// Example: countPairsWithDiffK([1, 5, 3, 4, 2], 2) -> 3  (Pairs: (1,3), (5,3), (4,2))
```
