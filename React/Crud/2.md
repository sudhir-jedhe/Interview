Here is a complete, lightweight CRUD (Create, Read, Update, Delete) React application built using React hooks (`useState`) and Tailwind CSS.

It manages a list of items (in this example, inventory/products) with full inline editing, creation modal/forms, and deletion features.

---

### Complete Single-File Component

```jsx
const INITIAL_DATA = [
  { id: 1, name: 'MacBook Pro 16"', category: 'Electronics', price: 2499, stock: 12 },

  { id: 3, name: 'Wireless Headphones', category: 'Electronics', price: 199, stock: 40 },
];

export default function CrudApp() {
  const [items, setItems] = useState(INITIAL_DATA);
```

### How Each CRUD Operation Works

1. **Create (`handleCreate`)**: Appends a new item object with a unique timestamp ID to the `items` state array and resets the form.
2. **Read (`filteredItems`)**: Displays state inside a table, automatically re-filtering based on the search query input.
3. **Update (`handleSaveUpdate`)**: Switches the specific row into an inline editable state using `editingId` and maps over `items` to merge changes on save.
4. **Delete (`handleDelete`)**: Filters out the item matching the clicked ID.

How do I connect this CRUD React component to a REST API backend like Express/Node using Axios or Fetch?

To connect your CRUD React app to a Node.js/Express backend, you replace local `useState` modifications with asynchronous API requests using **`fetch`** or **`axios`**.

Here is how to set up both sides—the Express backend endpoints and the React API logic with loading and error handling.

---

### 1. Express / Node.js API Endpoints

Your Node/Express backend needs to expose endpoints for each CRUD operation and enable **CORS** so your React app (running on a different port like `5173` or `3000`) can make requests to it.

```javascript
// server.js
const express = require('express');
const cors = require('cors');

const app = express();
app.use(cors()); // Allow requests from React frontend
app.use(express.json()); // Parse JSON bodies

let items = [
  { id: 1, name: 'MacBook Pro 16"', category: 'Electronics', price: 2499, stock: 12 },
  { id: 2, name: 'Ergonomic Chair', category: 'Furniture', price: 350, stock: 25 },
];

// READ: Get all items
app.get('/api/items', (req, res) => {
  res.json(items);
});

// CREATE: Add new item
app.post('/api/items', (req, res) => {
  const newItem = { id: Date.now(), ...req.body };
  items.unshift(newItem);
  res.status(201).json(newItem);
});

// UPDATE: Modify existing item
app.put('/api/items/:id', (req, res) => {
  const id = Number(req.params.id);
  items = items.map((item) => (item.id === id ? { ...item, ...req.body } : item));
  const updatedItem = items.find((item) => item.id === id);
  res.json(updatedItem);
});

// DELETE: Remove item
app.delete('/api/items/:id', (req, res) => {
  const id = Number(req.params.id);
  items = items.filter((item) => item.id !== id);
  res.json({ success: true, id });
});

app.listen(5000, () => console.log('Server running on port 5000'));

```

### 2. React CRUD Integration

Below is the refactored React logic replacing local mutations with API calls. You can choose either the **Fetch API** or **Axios** approach—both perform the exact same HTTP requests.

#### Option A: Using Standard `fetch`

```jsx
import React, { useState, useEffect } from 'react';

const API_URL = 'http://localhost:5000/api/items';

export default function CrudApp() {
  const [items, setItems] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  // --- 1. READ (GET) ---
  const fetchItems = async () => {
    try {
      setLoading(true);
      const res = await fetch(API_URL);
      if (!res.ok) throw new Error('Failed to fetch data');
      const data = await res.json();
      setItems(data);
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchItems();
  }, []);

  // --- 2. CREATE (POST) ---
  const handleCreate = async (formData) => {
    try {
      const res = await fetch(API_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(formData),
      });
      if (!res.ok) throw new Error('Failed to create item');
      const newItem = await res.json();
      setItems((prev) => [newItem, ...prev]);
    } catch (err) {
      alert(err.message);
    }
  };

  // --- 3. UPDATE (PUT) ---
  const handleSaveUpdate = async (id, updatedData) => {
    try {
      const res = await fetch(`${API_URL}/${id}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(updatedData),
      });
      if (!res.ok) throw new Error('Failed to update item');
      const updatedItem = await res.json();
      setItems((prev) =>
        prev.map((item) => (item.id === id ? updatedItem : item))
      );
    } catch (err) {
      alert(err.message);
    }
  };

  // --- 4. DELETE (DELETE) ---
  const handleDelete = async (id) => {
    try {
      const res = await fetch(`${API_URL}/${id}`, { method: 'DELETE' });
      if (!res.ok) throw new Error('Failed to delete item');
      setItems((prev) => prev.filter((item) => item.id !== id));
    } catch (err) {
      alert(err.message);
    }
  };

  if (loading) return <div>Loading items...</div>;
  if (error) return <div>Error: {error}</div>;

  return (
    <div>
      {/* Render UI table and forms passing handleCreate, handleSaveUpdate, handleDelete */}
    </div>
  );
}

```

### Best Practices to Keep in Mind

1. **Optimistic Updates vs. Pessimistic Updates**:

* **Pessimistic** (shown above): Wait for the server response before updating UI state. Safer and guarantees data integrity.
* **Optimistic**: Update React state immediately, then send request to backend. If the API call fails, revert state. Best for fast perceived performance.

1. **Centralized API Client**: As your app grows, move API calls into a separate service file (e.g., `api/itemsApi.js`) rather than putting `fetch`/`axios` directly inside your visual components.
2. **Data Fetching Libraries**: For real-world production apps, consider **TanStack Query (React Query)** or **SWR**. They handle automatic caching, refetching on window focus, loading states, and error retries out of the box without manual `useEffect` boilerplates.

How do I make this CRUD app persist its data using browser LocalStorage in React?

Persisting your React CRUD data to browser `localStorage` requires two primary mechanics:

1. **Reading initial state** from `localStorage` using a lazy initializer function inside `useState`.
2. **Writing updated state** to `localStorage` inside a `useEffect` hook whenever the `items` array changes.

---

### Refactored State Logic with LocalStorage

Here is how you update your React state to sync seamlessly with local storage:

```jsx
import React, { useState, useEffect } from 'react';

const STORAGE_KEY = 'crud_inventory_items';

const DEFAULT_DATA = [
  { id: 1, name: 'MacBook Pro 16"', category: 'Electronics', price: 2499, stock: 12 },
  { id: 2, name: 'Ergonomic Chair', category: 'Furniture', price: 350, stock: 25 },
];

export default function LocalStorageCrudApp() {
  // 1. READ: Lazy initial state function runs ONLY on initial component mount
  const [items, setItems] = useState(() => {
    try {
      const savedItems = localStorage.getItem(STORAGE_KEY);
      return savedItems ? JSON.parse(savedItems) : DEFAULT_DATA;
    } catch (error) {
      console.error('Failed to parse localStorage data:', error);
      return DEFAULT_DATA;
    }
  });

  // 2. WRITE: Syncs to localStorage whenever `items` changes (Create, Update, Delete)
  useEffect(() => {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(items));
    } catch (error) {
      console.error('Failed to save to localStorage:', error);
    }
  }, [items]);

  // --- CRUD Handlers (No changes required to your handler logic!) ---
  const handleCreate = (newItem) => {
    setItems((prev) => [newItem, ...prev]);
  };

  const handleUpdate = (id, updatedItem) => {
    setItems((prev) => prev.map((item) => (item.id === id ? updatedItem : item)));
  };

  const handleDelete = (id) => {
    setItems((prev) => prev.filter((item) => item.id !== id));
  };

  const handleResetToDefault = () => {
    localStorage.removeItem(STORAGE_KEY);
    setItems(DEFAULT_DATA);
  };

  return (
    <div className="p-4">
      {/* Your standard Table / Form UI */}
      <button 
        onClick={handleResetToDefault}
        className="px-3 py-1 bg-gray-200 text-gray-700 rounded text-sm hover:bg-gray-300"
      >
        Reset to Defaults
      </button>
    </div>
  );
}

```

### Custom Hook Pattern (`useLocalStorage`)

If you plan to persist other pieces of state across your app (e.g., user settings, active tab, themes), extract the logic into a reusable **custom hook**:

```jsx
import { useState, useEffect } from 'react';

export function useLocalStorage(key, initialValue) {
  const [storedValue, setStoredValue] = useState(() => {
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.error(`Error reading localStorage key "${key}":`, error);
      return initialValue;
    }
  });

  useEffect(() => {
    try {
      window.localStorage.setItem(key, JSON.stringify(storedValue));
    } catch (error) {
      console.error(`Error writing localStorage key "${key}":`, error);
    }
  }, [key, storedValue]);

  return [storedValue, setStoredValue];
}

```

#### Usage in component

```jsx
function CrudApp() {
  const [items, setItems] = useLocalStorage('crud_items', DEFAULT_DATA);
  // Now handleCreate, handleUpdate, handleDelete will automatically persist!
}

```

---

### Key Production Considerations

1. **Lazy State Initialization**: Passing a callback `useState(() => ...)` instead of `useState(JSON.parse(...))` ensures `localStorage.getItem` and `JSON.parse` only run **once** on page load, avoiding expensive synchronous reads during re-renders.
2. **Quota Exceeded Errors**: Browser `localStorage` holds roughly **5MB–10MB** of string data. Always wrap calls in `try...catch` blocks to prevent app crashes if storage is full or disabled (e.g., private browsing restrictions).
3. **Multi-Tab Syncing**: If a user opens your app in two browser tabs, changes in Tab A won't automatically update Tab B. You can listen to the native `window.addEventListener('storage', ...)` event to synchronize state across open tabs.

How do I add table sorting by columns and pagination to this CRUD React component?

Adding sorting and pagination to a CRUD table requires managing state for three things:

1. **Sort Configuration** (`key` column name and `direction` `'asc'` | `'desc'`).
2. **Pagination State** (`currentPage` and `itemsPerPage`).
3. **Derived Data Pipeline** (Search $\rightarrow$ Sort $\rightarrow$ Paginate).

Here is the complete implementation updating your existing CRUD table with clickable table headers, page controls, and items-per-page selectors.

---

### Complete Component with Sorting & Pagination

```jsx
import React, { useState, useMemo } from 'react';
import { 
  Plus, Edit2, Trash2, Check, X, Search, 
  ChevronUp, ChevronDown, ChevronsUpDown, 
  ChevronLeft, ChevronRight 
} from 'lucide-react';

const INITIAL_DATA = [
  { id: 1, name: 'MacBook Pro 16"', category: 'Electronics', price: 2499, stock: 12 },
  { id: 2, name: 'Ergonomic Chair', category: 'Furniture', price: 350, stock: 25 },
  { id: 3, name: 'Wireless Headphones', category: 'Electronics', price: 199, stock: 40 },
  { id: 4, name: 'Mechanical Keyboard', category: 'Electronics', price: 120, stock: 18 },
  { id: 5, name: 'Standing Desk', category: 'Furniture', price: 650, stock: 8 },
  { id: 6, name: '4K Monitor', category: 'Electronics', price: 450, stock: 15 },
  { id: 7, name: 'USB-C Dock', category: 'Electronics', price: 89, stock: 30 },
];

export default function PaginatedSortedCrudTable() {
  const [items, setItems] = useState(INITIAL_DATA);
  const [search, setSearch] = useState('');
  const [editingId, setEditingId] = useState(null);

  // Form States
  const [formData, setFormData] = useState({ name: '', category: '', price: '', stock: '' });
  const [editFormData, setEditFormData] = useState({ name: '', category: '', price: '', stock: '' });
  const [isAdding, setIsAdding] = useState(false);

  // --- 1. SORTING STATE ---
  const [sortConfig, setSortConfig] = useState({ key: 'name', direction: 'asc' });

  // --- 2. PAGINATION STATE ---
  const [currentPage, setCurrentPage] = useState(1);
  const [itemsPerPage, setItemsPerPage] = useState(5);

  // Handle Sort Click
  const handleSort = (key) => {
    let direction = 'asc';
    if (sortConfig.key === key && sortConfig.direction === 'asc') {
      direction = 'desc';
    }
    setSortConfig({ key, direction });
  };

  // Reset to Page 1 when Search or Items Per Page Changes
  const handleSearchChange = (e) => {
    setSearch(e.target.value);
    setCurrentPage(1);
  };

  const handleItemsPerPageChange = (e) => {
    setItemsPerPage(Number(e.target.value));
    setCurrentPage(1);
  };

  // --- 3. DERIVED DATA PIPELINE (Search -> Sort -> Paginate) ---
  const processedData = useMemo(() => {
    // Step A: Filter
    let filtered = items.filter(
      (item) =>
        item.name.toLowerCase().includes(search.toLowerCase()) ||
        item.category.toLowerCase().includes(search.toLowerCase())
    );

    // Step B: Sort
    if (sortConfig.key) {
      filtered.sort((a, b) => {
        const aVal = a[sortConfig.key];
        const bVal = b[sortConfig.key];

        if (typeof aVal === 'string') {
          return sortConfig.direction === 'asc'
            ? aVal.localeCompare(bVal)
            : bVal.localeCompare(aVal);
        }
        return sortConfig.direction === 'asc' ? aVal - bVal : bVal - aVal;
      });
    }

    // Step C: Paginate
    const totalPages = Math.ceil(filtered.length / itemsPerPage) || 1;
    const startIndex = (currentPage - 1) * itemsPerPage;
    const paginatedItems = filtered.slice(startIndex, startIndex + itemsPerPage);

    return {
      paginatedItems,
      totalCount: filtered.length,
      totalPages,
    };
  }, [items, search, sortConfig, currentPage, itemsPerPage]);

  // CRUD Handlers
  const handleCreate = (e) => {
    e.preventDefault();
    if (!formData.name || !formData.category) return;
    const newItem = {
      id: Date.now(),
      name: formData.name,
      category: formData.category,
      price: Number(formData.price) || 0,
      stock: Number(formData.stock) || 0,
    };
    setItems([newItem, ...items]);
    setFormData({ name: '', category: '', price: '', stock: '' });
    setIsAdding(false);
  };
```

How do I make this LocalStorage React hook sync data in real time across multiple open browser tabs using the storage event listener?
To sync data in real time across multiple browser tabs, you can listen to the browser's native **`storage` event**.

The `storage` event fires on `window` whenever a `localStorage` item is created, updated, or deleted **in another tab or window** belonging to the same origin. (Crucially, it does *not* fire in the tab that made the change, which naturally prevents infinite updates.)

---

### Enhanced `useLocalStorage` Hook with Multi-Tab Sync

Here is the updated custom hook with the `storage` event listener integrated:

```jsx
import { useState, useEffect, useCallback } from 'react';

export function useLocalStorage(key, initialValue) {
  // 1. Read initial value from localStorage or fallback
  const [storedValue, setStoredValue] = useState(() => {
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.error(`[useLocalStorage] Error reading key "${key}":`, error);
      return initialValue;
    }
  });

  // 2. Custom setter to update React state & localStorage simultaneously
  const setValue = useCallback(
    (value) => {
      try {
        setStoredValue((prevValue) => {
          // Allow value to be a function so we have same API as useState
          const valueToStore = value instanceof Function ? value(prevValue) : value;
          window.localStorage.setItem(key, JSON.stringify(valueToStore));
          return valueToStore;
        });
      } catch (error) {
        console.error(`[useLocalStorage] Error setting key "${key}":`, error);
      }
    },
    [key]
  );

  // 3. Listen for changes in OTHER tabs
  useEffect(() => {
    const handleStorageChange = (event) => {
      // Ignore storage events from other keys or other storage mechanisms (sessionStorage)
      if (event.key !== key || event.storageArea !== window.localStorage) {
        return;
      }

      try {
        if (event.newValue === null) {
          // Item was removed in another tab
          setStoredValue(initialValue);
        } else {
          // Item was updated in another tab
          setStoredValue(JSON.parse(event.newValue));
        }
      } catch (error) {
        console.error(`[useLocalStorage] Error parsing multi-tab update for key "${key}":`, error);
      }
    };

    // Attach listener
    window.addEventListener('storage', handleStorageChange);

    // Cleanup on unmount
    return () => {
      window.removeEventListener('storage', handleStorageChange);
    };
  }, [key, initialValue]);

  return [storedValue, setValue];
}

```

---

### How It Works

1. **Explicit Storage Writing**: Instead of triggering `localStorage.setItem` in a `useEffect` on state change, `setValue` writes to `localStorage` immediately when invoked. This keeps synchronous browser storage aligned with React's state update queue.
2. **Selective Filtering**: Inside `handleStorageChange`, we check `event.key === key`. This ensures the hook only re-renders if the specific key it is tracking changes—not when unrelated `localStorage` items change.
3. **Deletion Handling**: If `event.newValue` is `null` (meaning `localStorage.removeItem(key)` or `localStorage.clear()` was called in another tab), the state falls back to `initialValue`.

How do I make two separate components on the same page sync using a custom window event when LocalStorage changes?

The native browser `storage` event **only fires across different tabs/windows**—it deliberately skips the current tab that made the change.

To sync two disconnected components on the **same page** without using a global state library (like Redux or Zustand), you can dispatch a **CustomEvent** whenever `localStorage` is updated.

---

### The Pattern: `CustomEvent` Dispatcher

When Component A updates `localStorage`, it also emits a custom event on `window` (e.g., `'local-storage-update'`). Component B listens to that custom event alongside the native `'storage'` event, achieving both **same-tab** and **cross-tab** synchronization.

```
Component A ──(setItem)──> localStorage
    │                           ▲
    └──(dispatchEvent)──┐       │ (storage event)
                        ▼       │
                    Component B (Same Tab & Other Tabs)

```

---

### Custom Hook: `useSyncLocalStorage`

Here is a hook that handles both same-tab custom events and cross-tab storage events cleanly:

```jsx
import { useState, useEffect, useCallback } from 'react';

// Custom event name for same-tab updates
const CUSTOM_STORAGE_EVENT = 'on-local-storage-update';

/**
 * Dispatch a custom event on window so components on the same page can listen.
 */
function dispatchStorageEvent(key, newValue) {
  window.dispatchEvent(
    new CustomEvent(CUSTOM_STORAGE_EVENT, {
      detail: { key, newValue },
    })
  );
}

export function useSyncLocalStorage(key, initialValue) {
  // 1. Read initial state
  const [storedValue, setStoredValue] = useState(() => {
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.error(`Error reading localStorage key "${key}":`, error);
      return initialValue;
    }
  });

  // 2. Custom setter: Updates state, localStorage, AND dispatches CustomEvent
  const setValue = useCallback(
    (value) => {
      try {
        setStoredValue((prevValue) => {
          const valueToStore = value instanceof Function ? value(prevValue) : value;
          const serialized = JSON.stringify(valueToStore);

          // Update browser storage
          window.localStorage.setItem(key, serialized);

          // Notify other components on the SAME page
          dispatchStorageEvent(key, serialized);

          return valueToStore;
        });
      } catch (error) {
        console.error(`Error writing localStorage key "${key}":`, error);
      }
    },
    [key]
  );

  // 3. Listen to both Same-Tab CustomEvent and Cross-Tab storage Event
  useEffect(() => {
    // Handler for SAME TAB (CustomEvent)
    const handleSameTabChange = (event) => {
      const { key: eventKey, newValue } = event.detail;
      if (eventKey !== key) return;

      try {
        setStoredValue(newValue === null ? initialValue : JSON.parse(newValue));
      } catch (error) {
        console.error(`Error parsing same-tab update for key "${key}":`, error);
      }
    };

    // Handler for OTHER TABS (Native StorageEvent)
    const handleCrossTabChange = (event) => {
      if (event.key !== key || event.storageArea !== window.localStorage) return;

      try {
        setStoredValue(event.newValue === null ? initialValue : JSON.parse(event.newValue));
      } catch (error) {
        console.error(`Error parsing cross-tab update for key "${key}":`, error);
      }
    };

    window.addEventListener(CUSTOM_STORAGE_EVENT, handleSameTabChange);
    window.addEventListener('storage', handleCrossTabChange);

    return () => {
      window.removeEventListener(CUSTOM_STORAGE_EVENT, handleSameTabChange);
      window.removeEventListener('storage', handleCrossTabChange);
    };
  }, [key, initialValue]);

  return [storedValue, setValue];
}

```

### Key Advantages

1. **Zero External Dependencies**: Uses native browser event mechanisms (`CustomEvent` and `addEventListener`).

3. **No State Drilling or Context Wrapper**: You don't need to wrap components in a React Context Provider; any component can consume `useSyncLocalStorage('key')` independently.
