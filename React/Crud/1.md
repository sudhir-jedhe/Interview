```jsx
  const [editingId, setEditingId] = useState(null);
```

### Core Axios vs. Fetch Differences

| Feature                   | `Axios`                                       | Native `fetch()`                                                   |
| ------------------------- | --------------------------------------------- | ------------------------------------------------------------------ |
| **JSON Data Parsing**     | Transforms automatically (`response.data`)    | Requires manual `await res.json()` call                            |
| **HTTP Errors (4xx/5xx)** | Automatically throws error into `catch` block | Does **not** reject error status; requires checking `if (!res.ok)` |
| **Headers Serialization** | Automatically serializes objects to JSON      | Requires explicit `'Content-Type': 'application/json'` header      |
| **Installation**          | NPM dependency (`npm i axios`)                | Built into browsers out-of-the-box                                 |

```jsx
    } catch (err) {

    } finally {
```

### Key Architectural Concepts

1. **Debouncing Input (`setTimeout`)**: Prevents making an HTTP API request on every keystroke. Instead, it waits until the user pauses typing for `400ms`.
2. **Synchronized State Effects**: `useEffect` runs `fetchItems()` automatically whenever `currentPage`, `limit`, or `debouncedQuery` changes.
3. **Total Record Count**: REST APIs typically supply total record counts in response headers (e.g., `x-total-count` or `Content-Range`), enabling dynamic calculation of total pages (`Math.ceil(total / limit)`).

Syncing search and pagination state with URL search parameters (e.g., `?q=react&page=2&limit=5`) keeps your app's UI shareable, bookmarkable, and responsive to the browser's back/forward navigation.

In **React Router v6 / v7**, you manage this using the `useSearchParams` hook.

### Key React Router Setup (`useSearchParams`)

Instead of creating separate `useState` hooks for `searchQuery` and `currentPage`, the URL itself becomes your primary source of truth.

```jsx
import React, { useEffect, useState, useCallback } from 'react';
import { useSearchParams } from 'react-router-dom';
import axios from 'axios';

const API_URL = 'https://jsonplaceholder.typicode.com/todos';

export default function UrlSyncedCrudApp() {
  // 1. React Router Hook for URL query string management
  const [searchParams, setSearchParams] = useSearchParams();

  // 2. Read values directly from URL (with fallbacks)
  const queryParam = searchParams.get('q') || '';
  const pageParam = parseInt(searchParams.get('page') || '1', 10);
  const limitParam = parseInt(searchParams.get('limit') || '5', 10);

  // Local state for the text input (to support smooth typing & debouncing)
  const [searchInput, setSearchInput] = useState(queryParam);
  
  // Data & API state
  const [items, setItems] = useState([]);
  const [totalCount, setTotalCount] = useState(0);
  const [loading, setLoading] = useState(false);

  // Sync local input state if user navigates Back/Forward in the browser
  useEffect(() => {
    setSearchInput(queryParam);
  }, [queryParam]);

  // ---------------------------------------------------------
  // Debounce input changes -> Update URL Search Params
  // ---------------------------------------------------------
  useEffect(() => {
    const timer = setTimeout(() => {
      // Only update URL if the search text actually changed
      if (searchInput !== queryParam) {
        setSearchParams((prevParams) => {
          const newParams = new URLSearchParams(prevParams);
          if (searchInput.trim()) {
            newParams.set('q', searchInput.trim());
          } else {
            newParams.delete('q'); // Clean URL when search is empty
          }
          newParams.set('page', '1'); // Reset to page 1 on new search
          return newParams;
        });
      }
    }, 400);

    return () => clearTimeout(timer);
  }, [searchInput, queryParam, setSearchParams]);

  // ---------------------------------------------------------
  // Fetch API data driven strictly by URL query parameters
  // ---------------------------------------------------------
  const fetchItems = useCallback(async () => {
    setLoading(true);
    try {
      const response = await axios.get(API_URL, {
        params: {
          q: queryParam || undefined,
          _page: pageParam,
          _limit: limitParam,
        },
      });

      setItems(response.data);

      const totalHeader = response.headers['x-total-count'];
      setTotalCount(totalHeader ? parseInt(totalHeader, 10) : 200);
    } catch (err) {
      console.error('API Error:', err);
    } finally {
      setLoading(false);
    }
  }, [queryParam, pageParam, limitParam]);

  useEffect(() => {
    fetchItems();
  }, [fetchItems]);

  // ---------------------------------------------------------
  // URL Update Helpers for Pagination & Page Limit
  // ---------------------------------------------------------
  const handlePageChange = (newPage) => {
    setSearchParams((prevParams) => {
      const newParams = new URLSearchParams(prevParams);
      newParams.set('page', newPage.toString());
      return newParams;
    });
  };

  const handleLimitChange = (newLimit) => {
    setSearchParams((prevParams) => {
      const newParams = new URLSearchParams(prevParams);
      newParams.set('limit', newLimit.toString());
      newParams.set('page', '1'); // Reset to page 1 on limit change
      return newParams;
    });
  };

  const totalPages = Math.ceil(totalCount / limitParam) || 1;

  return (
    <div style={styles.container}>
      <h2>URL Synced CRUD App</h2>
      <p style={styles.subtext}>Try refreshing or sharing the URL after searching or changing pages!</p>

      {/* SEARCH & LIMIT CONTROLS */}
      <div style={styles.filterBar}>
        <input
          type="text"
          placeholder="Search items..."
          value={searchInput}
          onChange={(e) => setSearchInput(e.target.value)}
          style={styles.searchInput}
        />
        <select
          value={limitParam}
          onChange={(e) => handleLimitChange(e.target.value)}
          style={styles.select}
        >
          <option value={5}>5 per page</option>
          <option value={10}>10 per page</option>
          <option value={20}>20 per page</option>
        </select>
      </div>

      {/* DATA DISPLAY */}
      <div style={styles.list}>
        {loading ? (
          <p>Loading data for page {pageParam}...</p>
        ) : items.length === 0 ? (
          <p>No results found for "{queryParam}"</p>
        ) : (
          <ul style={{ listStyle: 'none', padding: 0 }}>
            {items.map((item) => (
              <li key={item.id} style={styles.listItem}>
                <span>{item.title}</span>
              </li>
            ))}
          </ul>
        )}
      </div>

      {/* PAGINATION FOOTER */}
      <div style={styles.pagination}>
        <button
          onClick={() => handlePageChange(pageParam - 1)}
          disabled={pageParam <= 1 || loading}
          style={styles.pageBtn}
        >
          &laquo; Prev
        </button>

        <span>
          Page <strong>{pageParam}</strong> of <strong>{totalPages}</strong>
        </span>

        <button
          onClick={() => handlePageChange(pageParam + 1)}
          disabled={pageParam >= totalPages || loading}
          style={styles.pageBtn}
        >
          Next &raquo;
        </button>
      </div>
    </div>
  );
}

const styles = {
  container: { maxWidth: '550px', margin: '20px auto', fontFamily: 'sans-serif' },
  subtext: { color: '#666', fontSize: '0.9em', marginTop: '-10px', marginBottom: '15px' },
  filterBar: { display: 'flex', gap: '10px', marginBottom: '15px' },
  searchInput: { flex: 1, padding: '8px', borderRadius: '4px', border: '1px solid #ccc' },
  select: { padding: '8px', borderRadius: '4px', border: '1px solid #ccc' },
  list: { padding: '15px', border: '1px solid #ccc', borderRadius: '8px', minHeight: '180px' },
  listItem: { padding: '8px 0', borderBottom: '1px solid #eee' },
  pagination: { display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginTop: '15px' },

};
```

### How This Works Behind the Scenes

1. **`useSearchParams()` replacing `useState**`: Reads parameters directly from the browser window URL (`window.location.search`).
2. **State-to-URL Updates**: Modifying query params is done via `setSearchParams(newParams)`, which pushes/replaces history entries without re-triggering full page reloads.
3. **Clean URL Handling**: Using `newParams.delete('q')` removes key-value pairs when the search query is empty so the URL stays clean (i.e., `?page=1` instead of `?q=&page=1`).
4. **Browser History Native Support**: Because parameters drive `useEffect`, using the browser **Back** and **Forward** buttons triggers automated re-fetching with previous parameters out-of-the-box.
