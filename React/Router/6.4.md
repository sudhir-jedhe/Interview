Here is a complete, production-ready React application showcasing **React Router v6.4+ Data APIs (`createBrowserRouter`)**.

It covers all key enterprise routing scenarios in a single file structure:

1. **Root & Layout Routing** with persistent `<Outlet/>` UI.
2. **Data Loaders & Actions** (parallel data fetching and mutation handling).
3. **Protected Routes & Role-Based Access Control (RBAC)** (Admin vs. User).
4. **Preserved Location Redirects** on login (`location.state`).
5. **Dynamic Route Parameters & Query Strings** (`useParams`, `useSearchParams`).
6. **Error Handling & 404 Boundaries** (`errorElement`, `useRouteError`).
7. **Lazy Loading & Code Splitting** (`React.lazy` + `Suspense`).

---

### Complete Code Setup

You can save this as `App.jsx` in a React project configured with React Router v6.4+ (`npm install react-router-dom`).

```jsx
import React, { createContext, useContext, useState, lazy, Suspense } from 'react';
import {
  createBrowserRouter,
  RouterProvider,
  Outlet,
  Link,
  NavLink,
  Navigate,
  useNavigate,
  useLocation,
  useParams,
  useSearchParams,
  useLoaderData,
  useRouteError,
  Form,
  redirect
} from 'react-router-dom';

// ============================================================================
// 1. MOCK AUTH & API UTILITIES
// ============================================================================

const AuthContext = createContext(null);

function AuthProvider({ children }) {
  // Mock logged-in user state: null | { name: string, role: 'User' | 'Admin' }
  const [user, setUser] = useState(null);

  const login = (username, role, callback) => {
    setUser({ name: username, role });
    if (callback) callback();
  };

  const logout = (callback) => {
    setUser(null);
    if (callback) callback();
  };

  return (
    <AuthContext.Provider value={{ user, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

const useAuth = () => useContext(AuthContext);

// Mock backend database
const mockProducts = [
  { id: '1', name: 'Laptop', price: '$999', category: 'tech' },
  { id: '2', name: 'Coffee Maker', price: '$49', category: 'home' },
  { id: '3', name: 'Headphones', price: '$199', category: 'tech' }
];

// ============================================================================
// 2. LAZY-LOADED COMPONENTS (Code Splitting Scenario)
// ============================================================================

// Lazy loading the Admin Panel
const AdminPanel = lazy(() =>
  Promise.resolve({
    default: function AdminPanel() {
      return (
        <div style={{ padding: '1rem', background: '#ffebee', borderRadius: '4px' }}>
          <h2>🔒 Admin Panel</h2>
          <p>Super secret controls! Only accessible by users with the "Admin" role.</p>
        </div>
      );
    }
  })
);

// ============================================================================
// 3. ROUTE GUARDS & ERROR BOUNDARIES
// ============================================================================

// Scenario: Protected Route & RBAC Guard
function ProtectedRoute({ allowedRoles }) {
  const { user } = useAuth();
  const location = useLocation();

  // 1. Not authenticated -> Redirect to /login and save attempted path
  if (!user) {
    return <Navigate to="/login" state={{ from: location }} replace />;
  }

  // 2. Lacks required role -> Redirect to /unauthorized
  if (allowedRoles && !allowedRoles.includes(user.role)) {
    return <Navigate to="/unauthorized" replace />;
  }

  // 3. Authorized -> Render nested routes
  return <Outlet />;
}

// Scenario: Global Error Boundary
function GlobalErrorElement() {
  const error = useRouteError();
  console.error(error);

  return (
    <div style={{ padding: '2rem', color: 'red' }}>
      <h1>Oops! Something went wrong.</h1>
      <p>{error.statusText || error.message || 'An unexpected error occurred.'}</p>
      <Link to="/">Return to Home</Link>
    </div>
  );
}

// ============================================================================
// 4. LAYOUTS & PAGES
// ============================================================================

// Main App Layout
function RootLayout() {
  const { user, logout } = useAuth();
  const navigate = useNavigate();

  return (
    <div style={{ fontFamily: 'sans-serif', maxWidth: '800px', margin: '0 auto', padding: '1rem' }}>
      <header style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', borderBottom: '1px solid #ccc', pb: '1rem' }}>
        <h2>React Router Master Application</h2>
        <div>
          {user ? (
            <span>
              Welcome, <strong>{user.name}</strong> ({user.role}) |{' '}
              <button onClick={() => logout(() => navigate('/'))}>Logout</button>
            </span>
          ) : (
            <Link to="/login">Login</Link>
          )}
        </div>
      </header>

      {/* Persistent Navigation Bar */}
      <nav style={{ display: 'flex', gap: '1rem', margin: '1rem 0' }}>
        <NavLink to="/" style={({ isActive }) => ({ fontWeight: isActive ? 'bold' : 'normal' })}>Home</NavLink>
        <NavLink to="/products" style={({ isActive }) => ({ fontWeight: isActive ? 'bold' : 'normal' })}>Products</NavLink>
        <NavLink to="/dashboard" style={({ isActive }) => ({ fontWeight: isActive ? 'bold' : 'normal' })}>Dashboard (User+)</NavLink>
        <NavLink to="/admin" style={({ isActive }) => ({ fontWeight: isActive ? 'bold' : 'normal' })}>Admin (Admin Only)</NavLink>
      </nav>

      <hr />

      {/* Dynamic Content Outlet */}
      <main style={{ marginTop: '1rem' }}>
        <Outlet />
      </main>
    </div>
  );
}

// Home Page
function HomePage() {
  return (
    <div>
      <h3>Home Page</h3>
      <p>Select a route from the navigation bar above to explore different routing patterns.</p>
    </div>
  );
}

// Login Page (Preserving Location Scenario)
function LoginPage() {
  const { login } = useAuth();
  const navigate = useNavigate();
  const location = useLocation();

  // Get path user tried to visit before being redirected to login
  const from = location.state?.from?.pathname || '/dashboard';

  const handleLogin = (role) => {
    login(`User_${Math.floor(Math.random() * 1000)}`, role, () => {
      // Send user back to where they originally intended to go
      navigate(from, { replace: true });
    });
  };

  return (
    <div>
      <h3>Login</h3>
      <p>Log in as a specific role to test access controls:</p>
      <div style={{ display: 'flex', gap: '1rem' }}>
        <button onClick={() => handleLogin('User')}>Login as Standard User</button>
        <button onClick={() => handleLogin('Admin')}>Login as Admin</button>
      </div>
    </div>
  );
}

// Scenario: Dynamic Route Parameters & Data Loaders
function ProductsPage() {
  const products = useLoaderData(); // Pre-fetched via router loader
  const [searchParams, setSearchParams] = useSearchParams();

  const categoryFilter = searchParams.get('category');

  const filteredProducts = categoryFilter
    ? products.filter((p) => p.category === categoryFilter)
    : products;

  return (
    <div>
      <h3>Products Catalog</h3>

      {/* Query Search Params controls */}
      <div style={{ marginBottom: '1rem', display: 'flex', gap: '0.5rem', alignItems: 'center' }}>
        <span>Filter Category:</span>
        <button onClick={() => setSearchParams({})}>All</button>
        <button onClick={() => setSearchParams({ category: 'tech' })}>Tech</button>
        <button onClick={() => setSearchParams({ category: 'home' })}>Home</button>
      </div>

      <ul>
        {filteredProducts.map((product) => (
          <li key={product.id}>
            <Link to={`/products/${product.id}`}>{product.name}</Link> - {product.price} ({product.category})
          </li>
        ))}
      </ul>
    </div>
  );
}

// Single Product Details
function ProductDetailsPage() {
  const { productId } = useParams(); // Extract :productId from URL
  const product = mockProducts.find((p) => p.id === productId);

  if (!product) {
    throw new Error('Product not found!'); // Triggers Route Error Boundary
  }

  return (
    <div>
      <h3>{product.name}</h3>
      <p><strong>Price:</strong> {product.price}</p>
      <p><strong>Category:</strong> {product.category}</p>
      <Link to="/products">← Back to Products</Link>
    </div>
  );
}

// User Dashboard
function DashboardPage() {
  return (
    <div>
      <h3>User Dashboard</h3>
      <p>This is a protected page accessible by both "User" and "Admin" roles.</p>
    </div>
  );
}

// 404 Not Found Page
function NotFoundPage() {
  return (
    <div>
      <h2>404 - Page Not Found</h2>
      <p>The page you are looking for does not exist.</p>
      <Link to="/">Go Home</Link>
    </div>
  );
}

function UnauthorizedPage() {
  return (
    <div style={{ color: 'darkred' }}>
      <h2>403 - Unauthorized</h2>
      <p>You do not have permission to view this page.</p>
      <Link to="/">Return to Home</Link>
    </div>
  );
}

// ============================================================================
// 5. ROUTER CONFIGURATION (`createBrowserRouter`)
// ============================================================================

const router = createBrowserRouter([
  {
    path: '/',
    element: <RootLayout />,
    errorElement: <GlobalErrorElement />,
    children: [
      // Public Routes
      { index: true, element: <HomePage /> },
      { path: 'login', element: <LoginPage /> },
      { path: 'unauthorized', element: <UnauthorizedPage /> },

      // Data Loader Scenario
      {
        path: 'products',
        element: <ProductsPage />,
        loader: async () => {
          // Simulate server network fetch delay
          await new Promise((resolve) => setTimeout(resolve, 200));
          return mockProducts;
        }
      },
      {
        path: 'products/:productId', // Dynamic param
        element: <ProductDetailsPage />
      },

      // Protected Routes Guard (User & Admin)
      {
        element: <ProtectedRoute allowedRoles={['User', 'Admin']} />,
        children: [
          { path: 'dashboard', element: <DashboardPage /> }
        ]
      },

      // Protected Routes Guard (Admin Only + Lazy Loading)
      {
        element: <ProtectedRoute allowedRoles={['Admin']} />,
        children: [
          {
            path: 'admin',
            element: (
              <Suspense fallback={<div>Loading Admin Module...</div>}>
                <AdminPanel />
              </Suspense>
            )
          }
        ]
      },

      // Catch-all 404 Route
      { path: '*', element: <NotFoundPage /> }
    ]
  }
]);

// ============================================================================
// 6. APPLICATION ENTRY POINT
// ============================================================================

export default function App() {
  return (
    <AuthProvider>
      <RouterProvider router={router} />
    </AuthProvider>
  );
}

```

React Router versioning skipped v7 as a traditional standalone rewrite—instead, **React Router v7 was released as the unified evolution of React Router v6 and Remix v2**, and **React Router v8** continues this unified framework trajectory.

Here is what you need to know about React Router v7 and v8, how the mental model has changed, and how to write routes in this modern architecture.

---

## Key Highlights of React Router v7 & v8

1. **Unification of React Router and Remix:**
React Router is no longer *just* a library for client-side matching. It is now a full-stack framework capability (formerly Remix) with support for SSR (Server-Side Rendering), SSG (Static Site Generation), and Server Actions out-of-the-box.
2. **First-Class Framework Capabilities:**
You can run React Router as a pure client SPA (`ssr: false`) or as a full-stack server application, using the exact same APIs (`loader`, `action`, `headers`, `meta`).
3. **Type-Safe Routes (`react-router` CLI):**
Automatic type generation for route params, loader data, and search params via the `./+types/` module generator.
4. **Simplification & Deprecations:**
Legacy APIs like `<Routes>` and `<Route>` nested manually inside component trees without a router provider are replaced by file-based or explicit route config trees (`routes.ts`).

---

## Modern React Router Setup (Framework Mode)

In modern React Router, routes are typically configured in a central `routes.ts` file or through file-based routing.

### 1. Central Route Configuration (`app/routes.ts`)

```typescript
// app/routes.ts
import { type RouteConfig, index, route, layout } from "@react-router/dev/routes";

export default [
  index("routes/home.tsx"),
  route("products", "routes/products.tsx"),
  route("products/:id", "routes/product-detail.tsx"),
  
  // Protected routes layout guard
  layout("routes/protected-layout.tsx", [
    route("dashboard", "routes/dashboard.tsx"),
    route("admin", "routes/admin.tsx"),
  ]),
] satisfies RouteConfig;

```

---

### 2. Full-Featured Route Component with Loaders & Actions (`app/routes/product-detail.tsx`)

Notice how `loader`, `action`, and the component share strict types provided by `./+types/product-detail`:

```tsx
// app/routes/product-detail.tsx
import { useLoaderData, Form, redirect } from "react-router";
import type { Route } from "./+types/product-detail";

// 1. Data Loader (Runs on server for SSR, or client fetch for SPA)
export async function loader({ params }: Route.LoaderArgs) {
  const res = await fetch(`https://api.example.com/products/${params.id}`);
  if (!res.ok) {
    throw new Response("Product Not Found", { status: 404 });
  }
  const product = await res.json();
  return { product };
}

// 2. Data Action (Handles Form Submissions / Mutations)
export async function action({ request, params }: Route.ActionArgs) {
  const formData = await request.formData();
  const intent = formData.get("intent");

  if (intent === "delete") {
    await fetch(`https://api.example.com/products/${params.id}`, { method: "DELETE" });
    return redirect("/products");
  }

  return null;
}

// 3. UI Component
export default function ProductDetail({ loaderData }: Route.ComponentProps) {
  const { product } = loaderData;

  return (
    <div>
      <h1>{product.name}</h1>
      <p>Price: ${product.price}</p>

      {/* React Router Form for server/client mutations */}
      <Form method="post">
        <button type="submit" name="intent" value="delete">
          Delete Product
        </button>
      </Form>
    </div>
  );
}

// 4. Built-in Error Boundary per route
export function ErrorBoundary({ error }: Route.ErrorBoundaryProps) {
  return (
    <div className="error-container">
      <h2>Something went wrong!</h2>
      <p>{error instanceof Error ? error.message : "Failed to load product"}</p>
    </div>
  );
}

```

## Comparison: v6 vs. v7/v8 Architecture

| Feature            | React Router v6                                | React Router v7 & v8                                     |
| ------------------ | ---------------------------------------------- | -------------------------------------------------------- |
| **Framework Role** | Client-side routing library                    | Unified Client + Server Framework                        |
| **Typesafety**     | Manual casting (`useParams<{ id: string }>()`) | Auto-generated route types (`./+types/...`)              |
| **Data Fetching**  | `useLoaderData()` with `createBrowserRouter`   | Native `loader` export + SSR hydration                   |
| **Mutations**      | Manual `fetch`/`axios` inside event handlers   | Native `action` export + `<Form>` components             |
| **Entry Point**    | `main.jsx` rendering `<RouterProvider>`        | Configured via `app/root.tsx` & `react-router.config.ts` |

---

## Migration Steps from v6 to v7/v8

1. **Enable v6 Future Flags:** Before upgrading, ensure your v6 app has all `future` flags enabled in `createBrowserRouter` (e.g., `v7_relativeSplatPath`, `v7_fetcherPersist`).
2. **Update Package Name:** Replace `react-router-dom` imports with `react-router`.
3. **Adopt Data APIs:** Replace `useEffect` fetching with `loader` functions and `useLoaderData()`.

How do I configure React Router v7/v8 as a client-side SPA without SSR?
To configure React Router v7/v8 as a traditional **Single Page Application (SPA)** with client-side rendering (CSR) and zero Server-Side Rendering (SSR), you set the `ssr: false` flag in your React Router configuration.

This allows you to keep all the modern features—data loaders, actions, type generation, and routing layout—while producing a static SPA bundle (HTML, JS, CSS) that can be hosted anywhere (S3, Vercel, Netlify, Nginx, GitHub Pages).

---

## Step 1: Set `ssr: false` in Config

In the root of your project, open or create `react-router.config.ts` and disable SSR:

```typescript
// react-router.config.ts
import type { Config } from "@react-router/dev/config";

export default {
  // Disable Server-Side Rendering (Builds a client-side SPA)
  ssr: false,
} satisfies Config;

```

## Step 2: Configure Client Entry Points

Ensure your application root and entry points are set up for client hydration/rendering.

### 1. Root Component (`app/root.tsx`)

In SPA mode, React Router needs a placeholder element to mount into while the app initializes on the client. Use `HydrateFallback` for initial loading UI.

```tsx
// app/root.tsx
import {
  Links,
  Meta,
  Outlet,
  Scripts,
  ScrollRestoration,
} from "react-router";

// Rendered while client-side JS bundle loads
export function HydrateFallback() {
  return (
    <div id="loading-screen">
      <p>Loading Application...</p>
    </div>
  );
}

export function Layout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en">
      <head>
        <meta charSet="utf-8" />
        <meta name="viewport" content="width=device-width, initial-scale=1" />
        <Meta />
        <Links />
      </head>
      <body>
        {children}
        <ScrollRestoration />
        <Scripts />
      </body>
    </html>
  );
}

export default function App() {
  return <Outlet />;
}

```

## Step 4: Use `clientLoader` and `clientAction` (Optional)

In pure SPA mode, standard `loader` and `action` functions still run on the client. However, if you explicitly want to signal that data fetching happens exclusively in the browser (e.g., accessing `window.localStorage`, `IndexedDB`, or client-side SDKs), use **`clientLoader`** and **`clientAction`**:

```tsx
// app/routes/dashboard.tsx
import { useLoaderData } from "react-router";
import type { Route } from "./+types/dashboard";

// Runs strictly in the browser
export async function clientLoader({ params }: Route.ClientLoaderArgs) {
  const token = localStorage.getItem("auth_token");
  
  const response = await fetch("https://api.example.com/user/dashboard", {
    headers: { Authorization: `Bearer ${token}` },
  });

  if (!response.ok) {
    throw new Error("Failed to fetch dashboard data");
  }

  return response.json();
}

export default function Dashboard({ loaderData }: Route.ComponentProps) {
  return (
    <div>
      <h1>User Dashboard</h1>
      <pre>{JSON.stringify(loaderData, null, 2)}</pre>
    </div>
  );
}

```

```bash
# Generate the client-side SPA build
npm run build

```

The output in your `build/client/` directory contains:

* Static JS and CSS assets

### Web Server Directives (Handling SPA Deep Links)

Since all routes are handled on the client, configure your web server/host to redirect all non-file URL requests back to `/index.html`:

* **Nginx:** `try_files $uri $uri/ /index.html;`
* **Netlify (`public/_redirects`):** `/*  /index.html  200`
* **Vercel (`vercel.json`):**

```json
{
  "rewrites": [{ "source": "/(.*)", "destination": "/index.html" }]
}

```
