```jsx
  const initialColors = Array(7).fill("yellow");
  const [colors, setColors] = useState(initialColors);
  const [clickedOrder, setClickedOrder] = useState([]);

  const handleClick = (index) => {
    let newColors = [...colors];
    newColors[index] = "green";
    setColors(newColors);
    setClickedOrder([...clickedOrder, index]);
  };

  const resetColors = () => {
    setTimeout(() => {
      let newColors = [...colors];
      let resetOrder = [...clickedOrder];
      const interval = setInterval(() => {
        if (resetOrder.length === 0) {
          clearInterval(interval);
          return;
        }
        const index = resetOrder.pop();
        newColors[index] = "yellow";
        setColors([...newColors]);
      }, 1000);
    }, 1000);
  };

  if (clickedOrder.length === 7) resetColors();

  return (
    <div
      style={{
        display: "flex",
        flexDirection: "row",
        flexWrap: "wrap",
        width: "300px",
      }}
    >
      {colors.map((color, index) => (
        <Box key={index} color={color} onClick={() => handleClick(index)} />
      ))}
    </div>
  );
```

### **Machine Coding 1: C-Shape Grid with Queue Unstacking**

#### **Key Requirements Handled:**

1. Renders 7 boxes specifically arranged in a **'C' shape** (3 top, 1 middle-left, 3 bottom).
2. Clicking a box turns it green and records click order.
3. Once all 7 boxes are green, they automatically revert to yellow **one by one every 1 second in the exact order they were clicked** (Queue behavior), disabling user clicks while unstacking.

```jsx
import React, { useState, useEffect, useRef } from "react";

const C_SHAPE_GRID = [
  [1, 1, 1],
  [1, 0, 0],
  [1, 1, 1],
];

export default function CShapeGrid() {
  const [order, setOrder] = useState([]);
  const [isDeactivating, setIsDeactivating] = useState(false);
  const timerRef = useRef(null);

  const totalBoxes = C_SHAPE_GRID.flat().filter(Boolean).length; // 7

  const handleBoxClick = (index) => {
    if (order.includes(index) || isDeactivating) return;
    setOrder((prev) => [...prev, index]);
  };

  useEffect(() => {
    if (order.length === totalBoxes && !isDeactivating) {
      setIsDeactivating(true);

      timerRef.current = setInterval(() => {
        setOrder((prevOrder) => {
          if (prevOrder.length === 0) {
            clearInterval(timerRef.current);
            setIsDeactivating(false);
            return [];
          }
          // Remove from start of array (FIFO order)
          return prevOrder.slice(1);
        });
      }, 1000);
    }

    return () => clearInterval(timerRef.current);
  }, [order.length, totalBoxes, isDeactivating]);

  let boxIndexCounter = 0;

  return (
    <div style={{ display: "grid", gridTemplateColumns: "repeat(3, 60px)", gap: "10px", margin: "20px" }}>
      {C_SHAPE_GRID.map((row, rowIndex) =>
        row.map((col, colIndex) => {
          if (col === 0) {
            return <div key={`${rowIndex}-${colIndex}`} />;
          }

          const currentBoxIndex = boxIndexCounter++;
          const isSelected = order.includes(currentBoxIndex);

          return (
            <button
              key={currentBoxIndex}
              onClick={() => handleBoxClick(currentBoxIndex)}
              disabled={isDeactivating}
              style={{
                width: "60px",
                height: "60px",
                backgroundColor: isSelected ? "green" : "yellow",
                border: "2px solid #333",
                borderRadius: "8px",
                cursor: isDeactivating ? "not-allowed" : "pointer",
                transition: "background-color 0.3s ease",
              }}
            />
          );
        })
      )}
    </div>
  );
}
```

### **Machine Coding 2: Table with Population API, Debouncing, Search, & Delete**

#### **Key Requirements Handled:**

1. Fetches data from Data USA public population API (replacing broken short link).
2. Displays Year and Population in a clean table.
3. Debounced input filtering on Year/Population without breaking input responsiveness.
4. **Reset Button** restores initial API data.
5. **Delete Option** removes specific rows from the displayed list.

```jsx
import React, { useState, useEffect, useMemo } from "react";

// Helper debounce function
function debounce(func, delay = 300) {
  let timer;
  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => func(...args), delay);
  };
}

export default function PopulationTable() {
  const [initialData, setInitialData] = useState([]);
  const [displayData, setDisplayData] = useState([]);
  const [searchTerm, setSearchTerm] = useState("");
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // Fetch data from public DataUSA Population API
    fetch("https://datausa.io/api/data?drilldowns=Nation&measures=Population")
      .then((res) => res.json())
      .then((res) => {
        const formattedData = res.data.map((item) => ({
          id: item["ID Year"] || item["Year"],
          year: String(item["Year"]),
          population: Number(item["Population"]),
        }));
        setInitialData(formattedData);
        setDisplayData(formattedData);
        setLoading(false);
      })
      .catch((err) => {
        console.error("Fetch error:", err);
        setLoading(false);
      });
  }, []);

  // Debounced search logic
  const debouncedFilter = useMemo(
    () =>
      debounce((query, data) => {
        if (!query.trim()) {
          setDisplayData(data);
          return;
        }
        const lowerQ = query.toLowerCase();
        const filtered = data.filter(
          (item) =>
            item.year.toLowerCase().includes(lowerQ) ||
            item.population.toString().includes(lowerQ)
        );
        setDisplayData(filtered);
      }, 400),
    []
  );

  const handleInputChange = (e) => {
    const value = e.target.value;
    setSearchTerm(value);
    debouncedFilter(value, initialData);
  };

  const handleReset = () => {
    setSearchTerm("");
    setDisplayData(initialData);
  };

  const handleDelete = (id) => {
    setDisplayData((prev) => prev.filter((item) => item.id !== id));
  };

  if (loading) return <div>Loading data...</div>;

  return (
    <div style={{ padding: "20px" }}>
      <div style={{ marginBottom: "15px", display: "flex", gap: "10px" }}>
        <input
          type="text"
          value={searchTerm}
          onChange={handleInputChange}
          placeholder="Search year or population..."
          style={{ padding: "8px", width: "250px" }}
        />
        <button onClick={handleReset} style={{ padding: "8px 16px" }}>
          Reset
        </button>
      </div>

      <table border="1" cellPadding="10" cellSpacing="0" style={{ width: "100%", maxWidth: "500px" }}>
        <thead>
          <tr style={{ backgroundColor: "#f2f2f2" }}>
            <th>Year</th>
            <th>Population</th>
            <th>Action</th>
          </tr>
        </thead>
        <tbody>
          {displayData.length > 0 ? (
            displayData.map((row) => (
              <tr key={row.id}>
                <td>{row.year}</td>
                <td>{row.population.toLocaleString()}</td>
                <td>
                  <button onClick={() => handleDelete(row.id)}>Delete</button>
                </td>
              </tr>
            ))
          ) : (
            <tr>
              <td colSpan="3" style={{ textAlign: "center" }}>
                No matching records found
              </td>
            </tr>
          )}
        </tbody>
      </table>
    </div>
  );
}
```

### **Machine Coding 3: Precision Timer with Pause & Reset**

#### **Key Requirements Handled:**

1. Starts from 0 and increments every second.
2. Toggle button switches label between "Start" and "Stop".
3. Resume functionality picks up from paused state.
4. Clean reset button resets timer to 0 and stops interval.

```jsx
import React, { useState, useEffect } from "react";

export default function Timer() {
  const [seconds, setSeconds] = useState(0);
  const [isActive, setIsActive] = useState(false);

  useEffect(() => {
    let interval = null;

    if (isActive) {
      interval = setInterval(() => {
        setSeconds((prev) => prev + 1);
      }, 1000);
    } else {
      clearInterval(interval);
    }

    return () => clearInterval(interval);
  }, [isActive]);

  const toggleTimer = () => {
    setIsActive((prev) => !prev);
  };

  const resetTimer = () => {
    setIsActive(false);
    setSeconds(0);

    <div style={{ padding: "20px", fontFamily: "sans-serif" }}>
      <h2>Timer: {seconds}s</h2>
      <div style={{ display: "flex", gap: "10px" }}>
        <button onClick={toggleTimer} style={{ padding: "8px 16px" }}>
          {isActive ? "Stop" : "Start"}
        </button>
        <button onClick={resetTimer} style={{ padding: "8px 16px" }}>
          Reset
        </button>
      </div>
    </div>
  );
```

```jsx
          style={{ flex: 1, padding: "8px" }}
        />
        <button type="submit">Add Todo</button>
      </form>

      {/* Filter Buttons */}
      <div style={{ display: "flex", gap: "10px", marginBottom: "15px" }}>
        {["ALL", "ACTIVE", "COMPLETED"].map((f) => (
          <button
            key={f}
            onClick={() => setFilter(f)}
            style={{
              fontWeight: filter === f ? "bold" : "normal",
              textDecoration: filter === f ? "underline" : "none",
            }}
          >
            {f}
          </button>
        ))}
      </div>

      {/* Todo List */}
      <ul style={{ listStyle: "none", padding: 0 }}>
        {filteredTodos.map((todo) => (
          <li key={todo.id} style={{ marginBottom: "8px" }}>
            <label style={{ cursor: "pointer" }}>
              <input
                type="checkbox"
                checked={todo.completed}
                onChange={() => toggleTodo(todo.id)}
                style={{ marginRight: "8px" }}
              />
              <span style={{ textDecoration: todo.completed ? "line-through" : "none" }}>
                {todo.text}
              </span>
            </label>
          </li>
        ))}
      </ul>
    </div>
  );
}
```
