### Core Data Grid Engine

```tsx
import React, {
  useState,
  useMemo,
  useCallback,
  useRef,
  useEffect,
  useReducer,
} from 'react';

export type SortDirection = 'asc' | 'desc' | null;

export interface Column<T = any> {
  key: string;
  header: string;
  width?: number;
  sortable?: boolean;
  filterable?: boolean;
  editable?: boolean;
  pivotable?: boolean;
  groupable?: boolean;
  aggregate?: 'sum' | 'avg' | 'min' | 'max' | 'count';
  render?: (value: any, row: T, isEditing: boolean) => React.ReactNode;
  editor?: (
    value: any,
    onChange: (val: any) => void,
    onCommit: () => void,
    row: T
  ) => React.ReactNode;
}

export interface GridRow {
  id: string;
  children?: GridRow[];
  [key: string]: any;
}

export type ViewMode = 'grid' | 'list' | 'pivot';

export interface PivotSummary {
  [pivotGroupValue: string]: {
    [key: string]: number | string;
  };
}

export interface ServerSideHandlers {
  onFetchData?: (params: {
    page: number;
    pageSize: number;
    sortKey: string | null;
    sortDir: SortDirection;
    search: string;
    filters: Record<string, string>;
  }) => Promise<{ rows: GridRow[]; totalCount: number }>;
}

export interface GridProps {
  columns: Column[];
  data: GridRow[];
  pageSize?: number;
  rowHeight?: number;
  viewportHeight?: number;
  locale?: 'en' | 'es' | 'fr';
  serverSide?: ServerSideHandlers;
  totalServerRows?: number;
}

// Undo / Redo Reducer
interface HistoryState {
  past: GridRow[][];
  present: GridRow[];
  future: GridRow[][];
}

type HistoryAction =
  | { type: 'UPDATE'; payload: GridRow[] }
  | { type: 'UNDO' }
  | { type: 'REDO' };

function historyReducer(state: HistoryState, action: HistoryAction): HistoryState {
  switch (action.type) {
    case 'UPDATE':
      return {
        past: [...state.past, state.present],
        present: action.payload,
        future: [],
      };
    case 'UNDO':
      if (state.past.length === 0) return state;
      const previous = state.past[state.past.length - 1];
      const newPast = state.past.slice(0, state.past.length - 1);
      return {
        past: newPast,
        present: previous,
        future: [state.present, ...state.future],
      };
    case 'REDO':
      if (state.future.length === 0) return state;
      const next = state.future[0];
      const newFuture = state.future.slice(1);
      return {
        past: [...state.past, state.present],
        present: next,
        future: newFuture,
      };
    default:
      return state;
  }
}

export const DataGrid: React.FC<GridProps> = ({
  columns,
  data,
  pageSize = 10,
  rowHeight = 40,
  viewportHeight = 360,
  locale = 'en',
  serverSide,
  totalServerRows,
}) => {
  const t = LOCALES[locale] || LOCALES.en;

  // --- Undo/Redo State ---
  const [history, dispatchHistory] = useReducer(historyReducer, {
    past: [],
    present: data,
    future: [],
  });
  const currentData = history.present;

  useEffect(() => {
    dispatchHistory({ type: 'UPDATE', payload: data });
  }, [data]);

  // --- View Mode & Controls ---
  const [viewMode, setViewMode] = useState<ViewMode>('grid');
  const [globalSearch, setGlobalSearch] = useState('');
  const [columnFilters, setColumnFilters] = useState<Record<string, string>>({});
  const [sortKey, setSortKey] = useState<string | null>(null);
  const [sortDir, setSortDir] = useState<SortDirection>(null);

  // --- Pagination ---
  const [page, setPage] = useState(1);

  // --- Selection ---
  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());

  // --- Tree & Group Expansion ---
  const [expandedRowIds, setExpandedRowIds] = useState<Set<string>>(new Set());
  const [groupBy, setGroupBy] = useState<string | null>(null);

  // --- Inline Editing & Active Focus ---
  const [editingCell, setEditingCell] = useState<{ rowId: string; colKey: string } | null>(null);
  const [activeCell, setActiveCell] = useState<{ rowId: string; colKey: string } | null>(null);

  // --- Virtualisation Scroll Ref ---
  const [scrollTop, setScrollTop] = useState(0);
  const viewportRef = useRef<HTMLDivElement | null>(null);

  // Sync Server-side Mode
  useEffect(() => {
    if (serverSide?.onFetchData) {
      serverSide.onFetchData({
        page,
        pageSize,
        sortKey,
        sortDir,
        search: globalSearch,
        filters: columnFilters,
      });
    }
  }, [page, pageSize, sortKey, sortDir, globalSearch, columnFilters, serverSide]);

  // Handle Edits
  const handleCellCommit = useCallback(
    (rowId: string, colKey: string, newValue: any) => {
      const updateData = (rows: GridRow[]): GridRow[] => {
        return rows.map((r) => {
          if (r.id === rowId) {
            return { ...r, [colKey]: newValue };
          }
          if (r.children) {
            return { ...r, children: updateData(r.children) };
          }
          return r;
        });
      };
      const updated = updateData(currentData);
      dispatchHistory({ type: 'UPDATE', payload: updated });
      setEditingCell(null);
    },
    [currentData]
  );

  // Handle Tree Toggle
  const toggleRowExpansion = (id: string) => {
    setExpandedRowIds((prev) => {
      const next = new Set(prev);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });
  };

  // Flatten Rows for Tree Data Representation
  const flattenRows = useCallback(
    (rows: GridRow[], level = 0): (GridRow & { _level?: number; _hasChildren?: boolean })[] => {
      let result: (GridRow & { _level?: number; _hasChildren?: boolean })[] = [];
      for (const row of rows) {
        const hasChildren = Boolean(row.children && row.children.length > 0);
        result.push({ ...row, _level: level, _hasChildren: hasChildren });
        if (hasChildren && expandedRowIds.has(row.id)) {
          result = result.concat(flattenRows(row.children!, level + 1));
        }
      }
      return result;
    },
    [expandedRowIds]
  );

  // Process Filtering and Sorting Locally
  const processedData = useMemo(() => {
    let rows = flattenRows(currentData);

    // Global Search
    if (globalSearch.trim()) {
      const query = globalSearch.toLowerCase();
      rows = rows.filter((r) =>
        columns.some((col) => String(r[col.key] ?? '').toLowerCase().includes(query))
      );
    }

    // Column Filters
    Object.keys(columnFilters).forEach((colKey) => {
      const filterVal = columnFilters[colKey]?.toLowerCase();
      if (filterVal) {
        rows = rows.filter((r) =>
          String(r[colKey] ?? '').toLowerCase().includes(filterVal)
        );
      }
    });

    // Sorting
    if (sortKey && sortDir) {
      rows = [...rows].sort((a, b) => {
        const valA = a[sortKey];
        const valB = b[sortKey];
        if (valA < valB) return sortDir === 'asc' ? -1 : 1;
        if (valA > valB) return sortDir === 'asc' ? 1 : -1;
        return 0;
      });
    }

    // Grouping Transformation
    if (groupBy) {
      const groups: Record<string, GridRow[]> = {};
      rows.forEach((row) => {
        const groupVal = String(row[groupBy] ?? 'Unassigned');
        if (!groups[groupVal]) groups[groupVal] = [];
        groups[groupVal].push(row);
      });

      const groupedResult: GridRow[] = [];
      Object.keys(groups).forEach((gKey) => {
        const groupId = `group-${gKey}`;
        groupedResult.push({
          id: groupId,
          [groupBy]: `Group: ${gKey} (${groups[gKey].length})`,
          children: groups[gKey],
          _isGroupHeader: true,
        });
        if (expandedRowIds.has(groupId)) {
          groupedResult.push(...groups[gKey]);
        }
      });
      return groupedResult;
    }

    return rows;
  }, [currentData, flattenRows, globalSearch, columnFilters, sortKey, sortDir, groupBy, columns, expandedRowIds]);

  // Pagination Slice
  const totalRows = serverSide ? totalServerRows || 0 : processedData.length;
  const totalPages = Math.ceil(totalRows / pageSize) || 1;

  const paginatedData = useMemo(() => {
    if (serverSide) return processedData;
    const start = (page - 1) * pageSize;
    return processedData.slice(start, start + pageSize);
  }, [processedData, page, pageSize, serverSide]);

  // Pivot Calculation
  const pivotData = useMemo<PivotSummary>(() => {
    if (viewMode !== 'pivot') return {};
    const pivotCol = columns.find((c) => c.pivotable);
    if (!pivotCol) return {};

    const summary: PivotSummary = {};
    processedData.forEach((row) => {
      const pVal = String(row[pivotCol.key] ?? 'Other');
      if (!summary[pVal]) summary[pVal] = {};

      columns.forEach((c) => {
        if (c.aggregate && typeof row[c.key] === 'number') {
          const curr = (summary[pVal][c.key] as number) || 0;
          summary[pVal][c.key] = curr + row[c.key];
        }
      });
    });

    return summary;
  }, [viewMode, columns, processedData]);

  // Virtualisation Calculations
  const totalHeight = paginatedData.length * rowHeight;
  const visibleCount = Math.ceil(viewportHeight / rowHeight);
  const startIndex = Math.max(0, Math.floor(scrollTop / rowHeight) - 2);
  const endIndex = Math.min(paginatedData.length, startIndex + visibleCount + 4);
  const visibleRows = paginatedData.slice(startIndex, endIndex);

  // Keyboard Navigation & Clipboard Handling
  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (!activeCell) return;
    const colKeys = columns.map((c) => c.key);
    const currRowIdx = paginatedData.findIndex((r) => r.id === activeCell.rowId);
    const currColIdx = colKeys.indexOf(activeCell.colKey);

    if (e.key === 'ArrowDown' && currRowIdx < paginatedData.length - 1) {
      setActiveCell({ rowId: paginatedData[currRowIdx + 1].id, colKey: activeCell.colKey });
    } else if (e.key === 'ArrowUp' && currRowIdx > 0) {
      setActiveCell({ rowId: paginatedData[currRowIdx - 1].id, colKey: activeCell.colKey });
    } else if (e.key === 'ArrowRight' && currColIdx < colKeys.length - 1) {
      setActiveCell({ rowId: activeCell.rowId, colKey: colKeys[currColIdx + 1] });
    } else if (e.key === 'ArrowLeft' && currColIdx > 0) {
      setActiveCell({ rowId: activeCell.rowId, colKey: colKeys[currColIdx - 1] });
    } else if (e.key === 'Enter') {
      if (editingCell) {
        setEditingCell(null);
      } else {
        setEditingCell(activeCell);
      }
    } else if (e.ctrlKey || e.metaKey) {
      if (e.key === 'c') {
        // Copy cell to clipboard
        const val = paginatedData[currRowIdx]?.[activeCell.colKey];
        navigator.clipboard.writeText(String(val ?? ''));
      } else if (e.key === 'v') {
        // Paste from clipboard
        navigator.clipboard.readText().then((text) => {
          handleCellCommit(activeCell.rowId, activeCell.colKey, text);
        });
      } else if (e.key === 'z') {
        dispatchHistory({ type: 'UNDO' });
      } else if (e.key === 'y') {
        dispatchHistory({ type: 'REDO' });
      }
    }
  };

  // Export CSV
  const handleExportCSV = () => {
    const headers = columns.map((c) => `"${c.header}"`).join(',');
    const rows = processedData
      .map((r) => columns.map((c) => `"${String(r[c.key] ?? '').replace(/"/g, '""')}"`).join(','))
      .join('\n');
    const blob = new Blob([`${headers}\n${rows}`], { type: 'text/csv;charset=utf-8;' });
    const link = document.createElement('a');
    link.href = URL.createObjectURL(blob);
    link.setAttribute('download', `export_${Date.now()}.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };

  // Aggregation Calculations
  const aggregations = useMemo(() => {
    const res: Record<string, number> = {};
    columns.forEach((c) => {
      if (c.aggregate) {
        const nums = processedData
          .map((r) => Number(r[c.key]))
          .filter((n) => !isNaN(n));
        if (c.aggregate === 'sum') res[c.key] = nums.reduce((a, b) => a + b, 0);
        if (c.aggregate === 'avg') res[c.key] = nums.length ? nums.reduce((a, b) => a + b, 0) / nums.length : 0;
        if (c.aggregate === 'min') res[c.key] = Math.min(...nums);
        if (c.aggregate === 'max') res[c.key] = Math.max(...nums);
        if (c.aggregate === 'count') res[c.key] = nums.length;
      }
    });
    return res;
  }, [columns, processedData]);

  return (
    <div
      tabIndex={0}
      onKeyDown={handleKeyDown}
      style={{
        fontFamily: 'system-ui, sans-serif',
        border: '1px solid #d1d5db',
        borderRadius: '8px',
        background: '#ffffff',
        outline: 'none',
      }}
    >
      {/* --- Control Toolbar --- */}
      <div style={{ padding: '12px', borderBottom: '1px solid #e5e7eb', display: 'flex', gap: '8px', flexWrap: 'wrap', alignItems: 'center' }}>
        <input
          type="text"
          placeholder={t.search}
          value={globalSearch}
          onChange={(e) => setGlobalSearch(e.target.value)}
          style={{ padding: '6px 12px', borderRadius: '4px', border: '1px solid #d1d5db' }}
        />

        <button onClick={() => dispatchHistory({ type: 'UNDO' })} style={btnStyle}>{t.undo}</button>
        <button onClick={() => dispatchHistory({ type: 'REDO' })} style={btnStyle}>{t.redo}</button>
        <button onClick={handleExportCSV} style={{ ...btnStyle, background: '#10b981', color: '#fff' }}>{t.exportCsv}</button>

        <select
          value={viewMode}
          onChange={(e) => setViewMode(e.target.value as ViewMode)}
          style={{ padding: '6px 12px', borderRadius: '4px', border: '1px solid #d1d5db' }}
        >
          <option value="grid">{t.viewGrid}</option>
          <option value="list">{t.viewList}</option>
          <option value="pivot">{t.viewPivot}</option>
        </select>

        <select
          onChange={(e) => setGroupBy(e.target.value || null)}
          style={{ padding: '6px 12px', borderRadius: '4px', border: '1px solid #d1d5db' }}
        >
          <option value="">No Grouping</option>
          {columns.filter((c) => c.groupable).map((c) => (
            <option key={c.key} value={c.key}>Group by {c.header}</option>
          ))}
        </select>
      </div>

      {/* --- View Mode Rendering --- */}
      {viewMode === 'pivot' ? (
        <div style={{ padding: '16px' }}>
          <h3>{t.viewPivot}</h3>
          <table style={{ width: '100%', borderCollapse: 'collapse' }}>
            <thead>
              <tr style={{ background: '#f3f4f6' }}>
                <th style={thStyle}>Group</th>
                {columns.filter((c) => c.aggregate).map((c) => (
                  <th key={c.key} style={thStyle}>{c.header} ({c.aggregate})</th>
                ))}
              </tr>
            </thead>
            <tbody>
              {Object.keys(pivotData).map((pGroup) => (
                <tr key={pGroup} style={{ borderBottom: '1px solid #e5e7eb' }}>
                  <td style={tdStyle}>{pGroup}</td>
                  {columns.filter((c) => c.aggregate).map((c) => (
                    <td key={c.key} style={tdStyle}>{pivotData[pGroup][c.key] ?? '-'}</td>
                  ))}
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      ) : viewMode === 'list' ? (
        <div style={{ padding: '16px', maxHeight: viewportHeight, overflowY: 'auto' }}>
          {visibleRows.map((row) => (
            <div key={row.id} style={{ padding: '12px', borderBottom: '1px solid #e5e7eb', marginBottom: '8px' }}>
              {columns.map((c) => (
                <div key={c.key}>
                  <strong>{c.header}: </strong>
                  {row[c.key]}
                </div>
              ))}
            </div>
          ))}
        </div>
      ) : (
        /* --- Grid Virtualised Table --- */
        <div
          ref={viewportRef}
          onScroll={(e) => setScrollTop(e.currentTarget.scrollTop)}
          style={{ height: viewportHeight, overflowY: 'auto', position: 'relative' }}
          role="grid"
          aria-multiselectable="true"
        >
          {/* Virtual Spacer */}
          <div style={{ height: totalHeight, width: '100%', position: 'relative' }}>
            <table
              style={{
                position: 'absolute',
                top: startIndex * rowHeight,
                left: 0,
                width: '100%',
                borderCollapse: 'collapse',
              }}
            >
              <thead>
                <tr style={{ background: '#f9fafb', borderBottom: '2px solid #e5e7eb' }}>
                  <th style={{ ...thStyle, width: 40 }}>
                    <input
                      type="checkbox"
                      onChange={(e) => {
                        if (e.target.checked) setSelectedIds(new Set(processedData.map((r) => r.id)));
                        else setSelectedIds(new Set());
                      }}
                    />
                  </th>
                  {columns.map((col) => (
                    <th key={col.key} style={{ ...thStyle, width: col.width || 150 }}>
                      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
                        <span
                          onClick={() => {
                            if (!col.sortable) return;
                            setSortKey(col.key);
                            setSortDir((prev) => (prev === 'asc' ? 'desc' : 'asc'));
                          }}
                          style={{ cursor: col.sortable ? 'pointer' : 'default' }}
                        >
                          {col.header} {sortKey === col.key ? (sortDir === 'asc' ? '▲' : '▼') : ''}
                        </span>
                      </div>
                      {col.filterable && (
                        <input
                          type="text"
                          style={{ width: '90%', fontSize: '11px', marginTop: '4px' }}
                          onChange={(e) =>
                            setColumnFilters((prev) => ({ ...prev, [col.key]: e.target.value }))
                          }
                        />
                      )}
                    </th>
                  ))}
                </tr>
              </thead>
              <tbody>
                {visibleRows.map((row) => {
                  const isSelected = selectedIds.has(row.id);
                  const isGroupHeader = row._isGroupHeader;

                  if (isGroupHeader) {
                    return (
                      <tr key={row.id} style={{ background: '#e5e7eb', height: rowHeight }}>
                        <td colSpan={columns.length + 1} style={{ padding: '0 12px', fontWeight: 'bold' }}>
                          <button
                            onClick={() => toggleRowExpansion(row.id)}
                            style={{ border: 'none', background: 'none', cursor: 'pointer' }}
                          >
                            {expandedRowIds.has(row.id) ? '▼' : '►'} {row[groupBy!]}
                          </button>
                        </td>
                      </tr>
                    );
                  }

                  return (
                    <tr
                      key={row.id}
                      role="row"
                      aria-selected={isSelected}
                      style={{
                        height: rowHeight,
                        borderBottom: '1px solid #e5e7eb',
                        background: isSelected ? '#eff6ff' : 'transparent',
                      }}
                    >
                      <td style={{ ...tdStyle, width: 40 }}>
                        <input
                          type="checkbox"
                          checked={isSelected}
                          onChange={() => {
                            const next = new Set(selectedIds);
                            if (next.has(row.id)) next.delete(row.id);
                            else next.add(row.id);
                            setSelectedIds(next);
                          }}
                        />
                      </td>
                      {columns.map((col) => {
                        const isEditing = editingCell?.rowId === row.id && editingCell?.colKey === col.key;
                        const isActive = activeCell?.rowId === row.id && activeCell?.colKey === col.key;

                        return (
                          <td
                            key={col.key}
                            role="gridcell"
                            onClick={() => setActiveCell({ rowId: row.id, colKey: col.key })}
                            onDoubleClick={() => col.editable && setEditingCell({ rowId: row.id, colKey: col.key })}
                            style={{
                              ...tdStyle,
                              paddingLeft: `${(row._level || 0) * 20 + 8}px`,
                              outline: isActive ? '2px solid #3b82f6' : 'none',
                            }}
                          >
                            {row._hasChildren && col.key === columns[0].key && (
                              <button
                                onClick={() => toggleRowExpansion(row.id)}
                                style={{ border: 'none', background: 'none', cursor: 'pointer', marginRight: 4 }}
                              >
                                {expandedRowIds.has(row.id) ? '▼' : '►'}
                              </button>
                            )}

                            {isEditing ? (
                              col.editor ? (
                                col.editor(
                                  row[col.key],
                                  (val) => handleCellCommit(row.id, col.key, val),
                                  () => setEditingCell(null),
                                  row
                                )
                              ) : (
                                <input
                                  autoFocus
                                  defaultValue={row[col.key]}
                                  onBlur={(e) => handleCellCommit(row.id, col.key, e.target.value)}
                                  onKeyDown={(e) => {
                                    if (e.key === 'Enter') handleCellCommit(row.id, col.key, e.currentTarget.value);
                                  }}
                                  style={{ width: '100%' }}
                                />
                              )
                            ) : col.render ? (
                              col.render(row[col.key], row, false)
                            ) : (
                              row[col.key]
                            )}
                          </td>
                        );
                      })}
                    </tr>
                  );
                })}
              </tbody>
            </table>
          </div>
        </div>
      )}

      {/* --- Aggregation Summary Row --- */}
      <div style={{ padding: '8px 12px', background: '#f3f4f6', borderTop: '1px solid #e5e7eb', fontSize: '12px' }}>
        <strong>{t.aggregate}: </strong>
        {Object.keys(aggregations).map((k) => (
          <span key={k} style={{ marginRight: '16px' }}>
            {columns.find((c) => c.key === k)?.header}: {aggregations[k]}
          </span>
        ))}
      </div>

      {/* --- Pagination Controls --- */}
      <div style={{ padding: '12px', borderTop: '1px solid #e5e7eb', display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
        <span>
          {selectedIds.size} {t.rowsSelected}
        </span>
        <div>
          <button disabled={page === 1} onClick={() => setPage((p) => p - 1)} style={btnStyle}>
            Previous
          </button>
          <span style={{ margin: '0 8px' }}>
            {t.page} {page} {t.of} {totalPages}
          </span>
          <button disabled={page === totalPages} onClick={() => setPage((p) => p + 1)} style={btnStyle}>
            Next
          </button>
        </div>
      </div>
    </div>
  );
};

// Styling Utilities
```
