Here is a complete, production-ready implementation of **Role-Based Routing, Authentication, Authorization, and Protected Routes** in **React (v6 / v7)** using `React Router`, Context API, and JWT handling patterns.

---

### Key Concepts

1. **Authentication (Auth)**: Checks *who* you are (Are you logged in?).
2. **Authorization (RBAC)**: Checks *what* you can do (Do you have the required `role` or `permissions`?).
3. **Protected Route**: A wrapper component that intercepts navigation and redirects unauthenticated or unauthorized users.

### 2. Auth Context & Provider (`src/context/AuthContext.tsx`)

Manages login state, token persistence, and logout handlers across the application.

```tsx
import React, { createContext, useContext, useState, useEffect } from 'react';
import { User, Role } from '../types/auth';

interface AuthContextType {
  user: User | null;
  token: string | null;
  isAuthenticated: boolean;
  isLoading: boolean;
  login: (token: string, user: User) => void;
  logout: () => void;
  hasRole: (roles: Role[]) => boolean;
  hasPermission: (permission: string) => boolean;
}

const AuthContext = createContext<AuthContextType | null>(null);

export const AuthProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [user, setUser] = useState<User | null>(null);
  const [token, setToken] = useState<string | null>(localStorage.getItem('token'));
  const [isLoading, setIsLoading] = useState<boolean>(true);

  useEffect(() => {
    // Hydrate user session from localStorage or token decoding on refresh
    const storedUser = localStorage.getItem('user');
    if (token && storedUser) {
      try {
        setUser(JSON.parse(storedUser));
      } catch {
        logout();
      }
    }
    setIsLoading(false);
  }, [token]);

  const login = (authToken: string, userData: User) => {
    localStorage.setItem('token', authToken);
    localStorage.setItem('user', JSON.stringify(userData));
    setToken(authToken);
    setUser(userData);
  };

  const logout = () => {
    localStorage.removeItem('token');
    localStorage.removeItem('user');
    setToken(null);
    setUser(null);
  };

  const hasRole = (allowedRoles: Role[]): boolean => {
    if (!user) return false;
    return allowedRoles.includes(user.role);
  };

  const hasPermission = (permission: string): boolean => {
    if (!user) return false;
    return user.permissions.includes(permission);
  };

  return (
    <AuthContext.Provider
      value={{
        user,
        token,
        isAuthenticated: !!user && !!token,
        isLoading,
        login,
        logout,
        hasRole,
        hasPermission,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => {
  const context = useContext(AuthContext);
  if (!context) throw new Error('useAuth must be used within an AuthProvider');
  return context;
};

```

### 3. Reusable Protected Route Guard (`src/routes/ProtectedRoute.tsx`)

This wrapper handles **Authentication**, **Role Authorization**, **Permission Validation**, and **Fallback Redirection**.

```tsx
import React from 'react';
import { Navigate, Outlet, useLocation } from 'react-router-dom';
import { useAuth } from '../context/AuthContext';
import { Role } from '../types/auth';

interface ProtectedRouteProps {
  allowedRoles?: Role[];
  requiredPermission?: string;
  redirectTo?: string;
}

export const ProtectedRoute: React.FC<ProtectedRouteProps> = ({
  allowedRoles,
  requiredPermission,
  redirectTo = '/login',
}) => {
  const { isAuthenticated, isLoading, hasRole, hasPermission } = useAuth();
  const location = useLocation();

  // 1. Show loading indicator while session hydrates from storage
  if (isLoading) {
    return (
      <div className="flex h-screen items-center justify-center">
        <p className="text-sm font-semibold text-slate-500">Authenticating session...</p>
      </div>
    );
  }

  // 2. Unauthenticated check -> Redirect to Login with location state for return trip
  if (!isAuthenticated) {
    return <Navigate to={redirectTo} state={{ from: location }} replace />;
  }

  // 3. Role Authorization check
  if (allowedRoles && !hasRole(allowedRoles)) {
    return <Navigate to="/unauthorized" replace />;
  }

  // 4. Permission Authorization check (optional granular check)
  if (requiredPermission && !hasPermission(requiredPermission)) {
    return <Navigate to="/unauthorized" replace />;
  }

  // 5. Authorized -> Render child routes
  return <Outlet />;
};

```

### 5. Application Router Setup (`src/App.tsx`)

Nested routes inside `<ProtectedRoute/>` inherit authentication and authorization rules cleanly.

```tsx
import React from 'react';
import { BrowserRouter, Routes, Route, Navigate, Link } from 'react-router-dom';
import { AuthProvider, useAuth } from './context/AuthContext';
import { ProtectedRoute } from './routes/ProtectedRoute';
import { Login } from './pages/Login';
import { Unauthorized } from './pages/Unauthorized';

// Placeholder Pages
const Dashboard = () => <PageLayout title="User Dashboard" role="USER, MANAGER, ADMIN" />;
const ManagerPanel = () => <PageLayout title="Manager Console" role="MANAGER, ADMIN" />;
const AdminSettings = () => <PageLayout title="Admin Control Center" role="ADMIN" />;

const PageLayout: React.FC<{ title: string; role: string }> = ({ title, role }) => {
  const { user, logout } = useAuth();
  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 p-8 font-sans">
      <header className="flex justify-between items-center border-b border-slate-800 pb-4 mb-6">
        <div>
          <h1 className="text-2xl font-bold">{title}</h1>
          <p className="text-xs text-slate-400">Allowed Roles: <strong className="text-indigo-400">{role}</strong></p>
        </div>
        <div className="flex items-center gap-4">
          <span className="text-xs font-semibold bg-slate-900 border border-slate-800 px-3 py-1.5 rounded-lg">
            Role: <strong className="text-amber-400">{user?.role}</strong>
          </span>
          <button
            onClick={logout}
            className="px-3 py-1.5 bg-red-600/20 hover:bg-red-600 text-red-400 hover:text-white text-xs font-bold rounded-lg transition"
          >
            Logout
          </button>
        </div>
      </header>

      {/* Navigation */}
      <nav className="flex gap-4 text-xs font-bold border-b border-slate-800 pb-4 mb-6">
        <Link to="/dashboard" className="hover:text-indigo-400">Dashboard</Link>
        <Link to="/manager" className="hover:text-indigo-400">Manager Panel</Link>
        <Link to="/admin" className="hover:text-indigo-400">Admin Settings</Link>
      </nav>
    </div>
  );
};

export default function App() {
  return (
    <BrowserRouter>
      <AuthProvider>
        <Routes>
          {/* Public Routes */}
          <Route path="/login" element={<Login />} />
          <Route path="/unauthorized" element={<Unauthorized />} />

          {/* 1. Authenticated Users Only (ALL Roles) */}
          <Route element={<ProtectedRoute />}>
            <Route path="/dashboard" element={<Dashboard />} />
          </Route>

          {/* 2. Manager & Admin Roles Only */}
          <Route element={<ProtectedRoute allowedRoles={['MANAGER', 'ADMIN']} />}>
            <Route path="/manager" element={<ManagerPanel />} />
          </Route>

          {/* 3. Admin Role Only */}
          <Route element={<ProtectedRoute allowedRoles={['ADMIN']} />}>
            <Route path="/admin" element={<AdminSettings />} />
          </Route>

          {/* Fallback Catch-All */}
          <Route path="*" element={<Navigate to="/dashboard" replace />} />
        </Routes>
      </AuthProvider>
    </BrowserRouter>
  );
}

```

### Route Authorization Summary Matrix

| Route        | Requires Login? | Allowed Roles              | Behaviour when Unauthorized  |
| ------------ | --------------- | -------------------------- | ---------------------------- |
| `/login`     | No              | Public                     | Accessible to everyone       |
| `/dashboard` | Yes             | `USER`, `MANAGER`, `ADMIN` | Redirects to `/login`        |
| `/manager`   | Yes             | `MANAGER`, `ADMIN`         | Redirects to `/unauthorized` |
| `/admin`     | Yes             | `ADMIN`                    | Redirects to `/unauthorized` |
