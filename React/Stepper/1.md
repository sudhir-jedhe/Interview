Here is a complete, production-ready, config-driven React Router Stepper implementation that covers all your requirements.

It uses **React Router v6**, **Zod**, **React Hook Form**, and **localStorage** persistence.

---

### 1. Types & Config Schema (`types.ts`)

```typescript
import { z } from 'zod';

export type UserRole = 'admin' | 'user' | 'manager';
export type Permission = 'edit_billing' | 'manage_users' | 'view_analytics';

export interface UserContextType {
  role: UserRole;
  permissions: Permission[];
  featureFlags: Record<string, boolean>;
}

export interface StepConfig {
  id: string;
  path: string; // e.g. "personal-info"
  title: string;
  schema: z.ZodSchema<any>;
  component: React.ComponentType<{
    onNext: () => void;
    onBack: () => void;
    isFirst: boolean;
    isLast: boolean;
  }>;
  // Dynamic Visibility Checks
  rolesAllowed?: UserRole[];
  permissionsRequired?: Permission[];
  featureFlagRequired?: string;
  shouldShow?: (formData: any) => boolean;
}

```

### 2. Global Stepper Context & Provider (`StepperContext.tsx`)

This context manages:

1. Merged form state across steps.
2. `localStorage` persistence and hydration.
3. Filtering visible steps based on user roles, permissions, feature flags, and current form data.
4. Route-guard validation to prevent direct URL jumping past uncompleted steps.

```tsx
import React, { createContext, useContext, useState, useEffect, useMemo } from 'react';
import { useNavigate, useLocation } from 'react-router-dom';
import { StepConfig, UserContextType } from './types';

interface StepperContextValue {
  formData: Record<string, any>;
  updateFormData: (data: Record<string, any>) => void;
  visibleSteps: StepConfig[];
  currentStepIndex: number;
  goToNextStep: () => void;
  goToPreviousStep: () => void;
  resetStepper: () => void;
  user: UserContextType;
}

const STORAGE_KEY = 'stepper_form_data_v1';

const StepperContext = createContext<StepperContextValue | null>(null);

export const StepperProvider: React.FC<{
  steps: StepConfig[];
  user: UserContextType;
  children: React.ReactNode;
}> = ({ steps, user, children }) => {
  const navigate = useNavigate();
  const location = useLocation();

  // 1. Persisted Form Data State
  const [formData, setFormData] = useState<Record<string, any>>(() => {
    try {
      const saved = localStorage.getItem(STORAGE_KEY);
      return saved ? JSON.parse(saved) : {};
    } catch {
      return {};
    }
  });

  // Sync to localStorage
  useEffect(() => {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(formData));
  }, [formData]);

  const updateFormData = (data: Record<string, any>) => {
    setFormData((prev) => ({ ...prev, ...data }));
  };

  const resetStepper = () => {
    localStorage.removeItem(STORAGE_KEY);
    setFormData({});
    navigate(`/stepper/${visibleSteps[0].path}`);
  };

  // 2. Compute Visible Steps based on Roles, Permissions, Feature Flags & Form Data
  const visibleSteps = useMemo(() => {
    return steps.filter((step) => {
      // Role Check
      if (step.rolesAllowed && !step.rolesAllowed.includes(user.role)) {
        return false;
      }
      // Permission Check
      if (
        step.permissionsRequired &&
        !step.permissionsRequired.every((p) => user.permissions.includes(p))
      ) {
        return false;
      }
      // Feature Flag Check
      if (
        step.featureFlagRequired &&
        !user.featureFlags[step.featureFlagRequired]
      ) {
        return false;
      }
      // Dynamic Condition Check
      if (step.shouldShow && !step.shouldShow(formData)) {
        return false;
      }
      return true;
    });
  }, [steps, user, formData]);

  // 3. Get Current Step Index from URL
  const currentPath = location.pathname.split('/').pop();
  const currentStepIndex = visibleSteps.findIndex((s) => s.path === currentPath);

  // 4. Route Guarding - Redirect if direct URL access attempts to skip uncompleted steps
  useEffect(() => {
    if (visibleSteps.length === 0) return;

    // If route doesn't match any visible step, redirect to first
    if (currentStepIndex === -1) {
      navigate(`/stepper/${visibleSteps[0].path}`, { replace: true });
      return;
    }

    // Verify all previous steps pass validation
    for (let i = 0; i < currentStepIndex; i++) {
      const prevStep = visibleSteps[i];
      const result = prevStep.schema.safeParse(formData);
      if (!result.success) {
        // Redirect to earliest incomplete step
        navigate(`/stepper/${prevStep.path}`, { replace: true });
        break;
      }
    }
  }, [location.pathname, visibleSteps, formData, currentStepIndex, navigate]);

  const goToNextStep = () => {
    if (currentStepIndex < visibleSteps.length - 1) {
      navigate(`/stepper/${visibleSteps[currentStepIndex + 1].path}`);
    }
  };

  const goToPreviousStep = () => {
    if (currentStepIndex > 0) {
      navigate(`/stepper/${visibleSteps[currentStepIndex - 1].path}`);
    }
  };

  return (
    <StepperContext.Provider
      value={{
        formData,
        updateFormData,
        visibleSteps,
        currentStepIndex,
        goToNextStep,
        goToPreviousStep,
        resetStepper,
        user,
      }}
    >
      {children}
    </StepperContext.Provider>
  );
};

export const useStepper = () => {
  const context = useContext(StepperContext);
  if (!context) throw new Error('useStepper must be used within a StepperProvider');
  return context;
};

```

### 3. Breadcrumb & Step Navigation UI (`StepperBreadcrumb.tsx`)

Generates steps automatically from `visibleSteps`.

```tsx
import React from 'react';
import { useStepper } from './StepperContext';

export const StepperBreadcrumb: React.FC = () => {
  const { visibleSteps, currentStepIndex } = useStepper();

  return (
    <nav className="mb-8">
      <ol className="flex items-center w-full text-sm font-medium text-center text-gray-500 border-b border-gray-200 pb-4">
        {visibleSteps.map((step, index) => {
          const isCompleted = index < currentStepIndex;
          const isCurrent = index === currentStepIndex;

          return (
            <li
              key={step.id}
              className={`flex items-center space-x-2.5 ${
                isCurrent
                  ? 'text-blue-600 font-bold'
                  : isCompleted
                  ? 'text-green-600 font-semibold'
                  : 'text-gray-400'
              }`}
            >
              <span
                className={`flex items-center justify-center w-8 h-8 rounded-full border ${
                  isCurrent
                    ? 'border-blue-600 bg-blue-50'
                    : isCompleted
                    ? 'border-green-600 bg-green-50'
                    : 'border-gray-300'
                }`}
              >
                {isCompleted ? '✓' : index + 1}
              </span>
              <span>{step.title}</span>
              {index < visibleSteps.length - 1 && <span className="mx-2 text-gray-300">/</span>}
            </li>
          );
        })}
      </ol>
    </nav>
  );
};

```

### 4. Generic Step Form Wrapper Component

A higher-order helper or wrapper integrating **React Hook Form** + **Zod** schema for each step.

```tsx
import React from 'react';
import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { z } from 'zod';
import { useStepper } from './StepperContext';

interface StepFormProps {
  schema: z.ZodSchema<any>;
  children: (methods: ReturnType<typeof useForm>) => React.ReactNode;
}

export const StepFormWrapper: React.FC<StepFormProps> = ({ schema, children }) => {
  const { formData, updateFormData, goToNextStep, goToPreviousStep, currentStepIndex, visibleSteps } =
    useStepper();

  const methods = useForm({
    resolver: zodResolver(schema),
    defaultValues: formData,
    mode: 'onTouched',
  });

  const onSubmit = (data: any) => {
    updateFormData(data);
    if (currentStepIndex === visibleSteps.length - 1) {
      alert('Form Complete! Final Data logged in console.');
      console.log('Final Submitted Data:', { ...formData, ...data });
    } else {
      goToNextStep();
    }
  };

  const isFirst = currentStepIndex === 0;
  const isLast = currentStepIndex === visibleSteps.length - 1;

  return (
    <form onSubmit={methods.handleSubmit(onSubmit)} className="space-y-6">
      {children(methods)}

      <div className="flex justify-between pt-6 border-t border-gray-200">
        <button
          type="button"
          onClick={goToPreviousStep}
          disabled={isFirst}
          className="px-4 py-2 bg-gray-200 text-gray-700 rounded-md disabled:opacity-50"
        >
          Previous
        </button>
        <button
          type="submit"
          className="px-6 py-2 bg-blue-600 text-white rounded-md hover:bg-blue-700"
        >
          {isLast ? 'Submit' : 'Next'}
        </button>
      </div>
    </form>
  );
};

```

```tsx
import React from 'react';
import { z } from 'zod';
import { StepFormWrapper } from './StepFormWrapper';

export const personalInfoSchema = z.object({
  fullName: z.string().min(2, 'Name is required'),
```

### 7. Router Setup & Layout (`App.tsx`)

```tsx
import React from 'react';
import { BrowserRouter, Routes, Route, Navigate, Outlet } from 'react-router-dom';
import { StepperProvider, useStepper } from './StepperContext';
import { StepperBreadcrumb } from './StepperBreadcrumb';
import { STEPPER_CONFIG } from './stepper.config';
import { UserContextType } from './types';

// Mock Logged In User Context
const mockUser: UserContextType = {
  role: 'admin',
  permissions: ['edit_billing'],
  featureFlags: {
    enable_billing_step: true,
  },
};

const StepperLayout: React.FC = () => {
  const { visibleSteps, resetStepper } = useStepper();

  return (
    <div className="max-w-3xl mx-auto p-8 bg-white rounded-xl shadow-md mt-10">
      <div className="flex justify-between items-center mb-6">
        <h1 className="text-2xl font-bold">Multi-Step Onboarding</h1>
        <button
          onClick={resetStepper}
          className="text-xs text-red-500 underline"
        >
          Reset Form
        </button>
      </div>

      <StepperBreadcrumb />

      {/* Render Current Active Step */}
      <Outlet />
    </div>
  );
};

export default function App() {
  return (
    <BrowserRouter>
      <StepperProvider steps={STEPPER_CONFIG} user={mockUser}>
        <Routes>
          <Route path="/stepper" element={<StepperLayout />}>
            {STEPPER_CONFIG.map((step) => {
              const Component = step.component;
              return (
                <Route
                  key={step.id}
                  path={step.path}
                  element={<Component />}
                />
              );
            })}
          </Route>

          {/* Catch-all redirect */}
          <Route path="*" element={<Navigate to="/stepper/personal-info" replace />} />
        </Routes>
      </StepperProvider>
    </BrowserRouter>
  );
}
```
