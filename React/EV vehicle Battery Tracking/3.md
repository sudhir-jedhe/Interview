### 1. Real-Time WebSocket/SSE Stream & Mock Telemetry Simulator (`src/hooks/useTelemetryStream.js`)

```javascript
import { useState, useEffect, useRef, useCallback } from 'react';

// Simulated WebSocket / SSE Stream Manager
export function useTelemetryStream(vehicleId, options = {}) {
  const [liveData, setLiveData] = useState(null);
  const [packetLogs, setPacketLogs] = useState([]);
  const [criticalAlarm, setCriticalAlarm] = useState(null);
  const [isConnected, setIsConnected] = useState(false);
  const audioContextRef = useRef(null);

  // Play audio chime for thermal runaway or critical alerts
  const triggerAlarmChime = useCallback(() => {
    try {
      if (!audioContextRef.current) {
        audioContextRef.current = new (window.AudioContext || window.webkitAudioContext)();
      }
      const ctx = audioContextRef.current;
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.type = 'sawtooth';
      osc.frequency.setValueAtTime(880, ctx.currentTime); // 880Hz warning
      osc.frequency.exponentialRampToValueAtTime(440, ctx.currentTime + 0.3);
      gain.gain.setValueAtTime(0.15, ctx.currentTime);
      gain.gain.linearRampToValueAtTime(0.01, ctx.currentTime + 0.3);
      osc.connect(gain);
      gain.connect(ctx.destination);
      osc.start();
      osc.stop(ctx.currentTime + 0.35);
    } catch (e) {
      console.warn('Audio alert unavailable', e);
    }
  }, []);

  useEffect(() => {
    setIsConnected(true);

    // Simulate incoming MQTT/SSE packets every 1.5 seconds
    const interval = setInterval(() => {
      const now = new Date();
      const timestamp = now.toTimeString().split(' ')[0];

      // Jitter calculation
      const tempDelta = (Math.random() * 2 - 0.8).toFixed(1);
      const simulatedTemp = Math.min(68, Math.max(28, 38 + parseFloat(tempDelta)));
      const simulatedVoltageImbalance = (Math.random() * 220).toFixed(0); // mV

      const packet = {
        vehicleId,
        timestamp,
        lat: 6.1350 + (Math.random() - 0.5) * 0.005,
        lng: 1.2150 + (Math.random() - 0.5) * 0.005,
        speed: Math.floor(Math.random() * 45),
        soc: Math.max(10, Math.floor(65 - (now.getSeconds() % 60) * 0.1)),
        packTemp: simulatedTemp,
        cellImbalanceMv: parseInt(simulatedVoltageImbalance, 10),
        rawHex: `0xAA5501${Math.floor(Math.random() * 0xffffff).toString(16).padStart(6, '0')}`,
      };

      // Thermal Runaway / Fault Detection Logic
      if (packet.packTemp >= 60 || packet.cellImbalanceMv >= 195) {
        const alarmPayload = {
          id: `ALM-${Date.now().toString().slice(-4)}`,
          title: packet.packTemp >= 60 ? 'THERMAL RUNAWAY RISK' : 'CELL VOLTAGE IMBALANCE',
          message: packet.packTemp >= 60 
            ? `Pack temperature reached ${packet.packTemp}°C (Threshold: 60°C)`
            : `Cell delta exceeded safe range: ${packet.cellImbalanceMv} mV (> 200mV)`,
          severity: 'CRITICAL',
          timestamp,
        };
        setCriticalAlarm(alarmPayload);
        triggerAlarmChime();
      }

      setLiveData(packet);
      setPacketLogs((prev) => [packet, ...prev.slice(0, 19)]); // Keep last 20 raw frames
    }, 1500);

    return () => {
      clearInterval(interval);
      setIsConnected(false);
    };
  }, [vehicleId, triggerAlarmChime]);

  return { liveData, packetLogs, criticalAlarm, clearAlarm: () => setCriticalAlarm(null), isConnected };
}
```

### 3. Battery Degradation Curve SVG Chart (`src/components/telemetry/DegradationChart.jsx`)

Plots recorded SOH against charge cycle counts alongside the projected retirement baseline ($80\%$ EOL threshold).

```jsx
import React from 'react';

export default function DegradationChart({ data = [], height = 200 }) {
  const width = 500;
  const svgHeight = 150;
  const maxCycles = 1200;
  const minSoh = 60;
  const maxSoh = 100;

  const getCoordinates = (cycle, soh) => {
    const x = (cycle / maxCycles) * (width - 40) + 30;
    const y = svgHeight - 20 - ((soh - minSoh) / (maxSoh - minSoh)) * (svgHeight - 40);
    return { x, y };
  };

  // Build projected line
  const projectedPath = data
    .map((d, i) => {
      const { x, y } = getCoordinates(d.cycle, d.projectedSoh);
      return `${i === 0 ? 'M' : 'L'} ${x} ${y}`;
    })
    .join(' ');

  // Build actual recorded points
  const recordedPoints = data.filter((d) => d.soh !== null);
  const actualPath = recordedPoints
    .map((d, i) => {
      const { x, y } = getCoordinates(d.cycle, d.soh);
      return `${i === 0 ? 'M' : 'L'} ${x} ${y}`;
    })
    .join(' ');

  // 80% Retirement Threshold Line
  const retirementY = svgHeight - 20 - ((80 - minSoh) / (maxSoh - minSoh)) * (svgHeight - 40);

  return (
    <div className="w-full flex flex-col justify-between" style={{ height }}>
      <div className="w-full h-40 relative">
        <svg viewBox={`0 0 ${width} ${svgHeight}`} className="w-full h-full overflow-visible">
          {/* 80% EOL Threshold Line */}
          <line
            x1="30"
            y1={retirementY}
            x2={width - 10}
            y2={retirementY}
            stroke="#f43f5e"
            strokeWidth="1.5"
            strokeDasharray="4 4"
          />
          <text x={width - 95} y={retirementY - 4} fill="#f43f5e" fontSize="9" fontWeight="bold">
            80% EOL Threshold
          </text>

          {/* Projected Trend Line */}
          <path d={projectedPath} fill="none" stroke="#94a3b8" strokeWidth="2" strokeDasharray="3 3" />

          {/* Actual Recorded Line */}
          <path d={actualPath} fill="none" stroke="#1e60f0" strokeWidth="2.5" strokeLinecap="round" />

          {/* Actual Recorded Dots */}
          {recordedPoints.map((pt) => {
            const { x, y } = getCoordinates(pt.cycle, pt.soh);
            return <circle key={pt.cycle} cx={x} cy={y} r="3.5" fill="#1e60f0" />;
          })}
        </svg>
      </div>

      <div className="flex justify-between items-center text-[10px] text-slate-400 border-t border-slate-100 pt-1 font-mono">
        <span>0 Cycles</span>
        <span>300 Cycles</span>
        <span>600 Cycles</span>
        <span>900 Cycles</span>
        <span>1200 Cycles</span>
      </div>
      <div className="flex items-center justify-center gap-4 text-[10px] text-slate-500 mt-1">
        <span className="flex items-center gap-1">
          <span className="w-3 h-0.5 bg-blue-600 inline-block"></span> Recorded SOH

          <span className="w-3 h-0.5 border-b border-slate-400 border-dashed inline-block"></span> Projected Degradation
```

### 6. CSV & PDF Automated Fleet Report Generator (`src/utils/reportExporter.js`)

```javascript
export function exportFleetCsvReport(vehicleData, trips = []) {
  const totalKm = vehicleData.odometer || 133;
  const kwhConsumed = (totalKm * 0.038).toFixed(2); // Avg 38 Wh/km for electric 2-wheelers
  const co2OffsetKg = (totalKm * 0.082).toFixed(2); // Avg 82g CO2 saved vs 125cc petrol engine

  const csvRows = [
    ['Spiro Fleet Telematics & Carbon Offset Report'],
    ['Generated On', new Date().toISOString()],
    ['Vehicle ID', vehicleData.id],
    ['Model', vehicleData.model],
    ['Battery Pack ID', vehicleData.batteryId],
    [],
    ['Key Metric', 'Value', 'Unit'],
    ['Odometer Reading', totalKm, 'km'],
    ['Total Energy Expended', kwhConsumed, 'kWh'],
    ['Net Carbon Offset', co2OffsetKg, 'kg CO2eq'],
    ['Current State of Health (SOH)', `${vehicleData.soh}%`, '%'],
    [],
    ['Timestamp', 'Latitude', 'Longitude', 'Speed (km/h)', 'SOC (%)'],
  ];

  trips.forEach((t) => {
    csvRows.push([t.timestamp, t.lat, t.lng, t.speed, `${t.soc}%`]);
  });

  const csvContent = 'data:text/csv;charset=utf-8,' + csvRows.map((e) => e.join(',')).join('\n');
  const encodedUri = encodeURI(csvContent);
  const link = document.createElement('a');
  link.setAttribute('href', encodedUri);
  link.setAttribute('download', `Spiro_Fleet_Report_${vehicleData.id}_${Date.now()}.csv`);
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
}
```

```jsx
import { useTelemetryStream } from '../hooks/useTelemetryStream';

import TripPlaybackBar from '../components/telemetry/TripPlaybackBar';
import RemoteOtaControls from '../components/telemetry/RemoteOtaControls';
import DegradationChart from '../components/telemetry/DegradationChart';
import SwapAndMaintenanceTabs from '../components/telemetry/SwapAndMaintenanceTabs';
import AuditTrailDrawer from '../components/telemetry/AuditTrailDrawer';
import { exportFleetCsvReport } from '../utils/reportExporter';
import { VEHICLE_TRIP_HISTORY, GEOFENCE_PERIMETERS, isPointInsidePolygon } from '../api/mockAdapter';
import { MOCK_DEGRADATION_CURVE } from '../api/fleetModulesData';

export default function VehicleInfo() {
  const { id } = useParams();
  const vehicleId = id || 'EC2832';
  const { data: vehicle, loading } = useApi(`/vehicles/${vehicleId}`);

  // Live WebSocket Simulation
  const { liveData, criticalAlarm, clearAlarm, isConnected } = useTelemetryStream(vehicleId);

  // Playback States
  const [playbackIndex, setPlaybackIndex] = useState(0);
  const [isPlaying, setIsPlaying] = useState(false);
  const [playbackSpeed, setPlaybackSpeed] = useState(1);
  const [latestAuditLog, setLatestAuditLog] = useState(null);

  if (loading || !vehicle) {
    return <div className="p-12 text-center text-slate-400">Loading Telematics & Sensors...</div>;
  }

  // Active coordinates: Prefer Live Stream over historical replay if not actively scrubbing
  const currentPoint = isPlaying
    ? VEHICLE_TRIP_HISTORY[playbackIndex]
    : liveData || { lat: vehicle.lat, lng: vehicle.lng, speed: vehicle.avgVelocity, soc: vehicle.soc };

  // Geofence check
  const currentGeofence = GEOFENCE_PERIMETERS[0];
  const isInside = isPointInsidePolygon([currentPoint.lat, currentPoint.lng], currentGeofence.coordinates);
  const isGeofenceAlert = !isInside;

  const handleCommandAudit = (cmdType) => {
    setLatestAuditLog({
      id: `LOG-${Date.now().toString().slice(-4)}`,
      timestamp: new Date().toISOString().replace('T', ' ').slice(0, 19),
      actor: 'Super Admin',
      action: `REMOTE_${cmdType}`,
      target: vehicle.id,
      details: `Executed ${cmdType} OTA operation via cloud dispatcher.`,
      status: 'SUCCESS',
    });
  };

  return (
    <div className="space-y-5">
      {/* CRITICAL THERMAL / SAFETY ALARM BANNER (Triggered via WebSocket) */}
      {criticalAlarm && (
        <div className="bg-rose-600 text-white p-4 rounded-xl shadow-lg border border-rose-700 flex items-center justify-between animate-pulse">
          <div className="flex items-center gap-3">
            <span className="text-2xl">🚨</span>
            <div>
              <h4 className="font-black text-sm tracking-wide">{criticalAlarm.title}</h4>
              <p className="text-xs text-rose-100">{criticalAlarm.message}</p>
            </div>
          </div>
          <button
            onClick={clearAlarm}
            className="px-3 py-1 bg-white text-rose-700 font-bold text-xs rounded hover:bg-rose-50 transition"
          >
            Acknowledge Alert
          </button>
        </div>
      )}

      {/* Top Header & Export Action */}
      <div className="flex flex-wrap items-center justify-between gap-3">
        <div className="flex items-center gap-3">
          <h2 className="text-xl font-bold text-slate-800">Vehicle Telemetry Hub</h2>
          <span className="flex items-center gap-1.5 text-xs px-2.5 py-0.5 rounded-full bg-slate-100 border border-slate-200">
            <span className={`w-2 h-2 rounded-full ${isConnected ? 'bg-emerald-500 animate-ping' : 'bg-slate-400'}`}></span>
            <span className="font-mono text-slate-600">{isConnected ? 'Live WebSocket Active' : 'Disconnected'}</span>
          </span>
        </div>

        <button
          onClick={() => exportFleetCsvReport(vehicle, VEHICLE_TRIP_HISTORY)}
          className="bg-blue-600 hover:bg-blue-700 text-white text-xs font-bold px-4 py-2 rounded-lg transition shadow-xs flex items-center gap-1.5"
        >
          <span>📥</span> Download ESG & Fleet Telemetry CSV
        </button>
      </div>

      {/* Vehicle Identity & Real-Time Sensors */}
      <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">
        <div className="lg:col-span-4 bg-white p-5 rounded-xl border border-slate-200 shadow-sm flex flex-col justify-between">
          <div className="flex justify-between items-start">
            <div>
              <span className="text-xs font-mono text-slate-400">VIN: {vehicle.id}</span>
              <h3 className="text-xl font-black text-slate-800">{vehicle.model}</h3>
            </div>
            <span className="text-xs px-2.5 py-0.5 rounded-full font-bold bg-blue-600 text-white">
              {vehicle.status}
            </span>
          </div>

          <div className="my-3 flex justify-center text-5xl">🛵</div>

          <div className="space-y-1 text-xs border-t border-slate-100 pt-3 text-slate-600">
            <p><span className="font-semibold text-slate-700">IMEI:</span> {vehicle.imei}</p>
            <p><span className="font-semibold text-slate-700">Battery Pack:</span> {vehicle.batteryId}</p>
            {liveData?.rawHex && (
              <p className="font-mono text-[10px] text-slate-400 truncate">Last Frame: {liveData.rawHex}</p>
            )}
          </div>
        </div>

        {/* 6 Metric KPI Blocks */}
        <div className="lg:col-span-8 grid grid-cols-1 sm:grid-cols-3 gap-3">
          <div className="bg-[#4361ee] text-white p-4 rounded-xl flex items-center justify-between">
            <div>
              <p className="text-xs text-blue-200 font-semibold">Live Speed</p>
              <p className="text-3xl font-black mt-1">{currentPoint.speed} <span className="text-sm font-normal">km/h</span></p>
            </div>
            <span className="text-2xl">⏱️</span>
          </div>

          <div className="bg-[#4361ee] text-white p-4 rounded-xl flex items-center justify-between">
            <div>
              <p className="text-xs text-blue-200 font-semibold">Battery SOC</p>
              <p className="text-3xl font-black mt-1">{currentPoint.soc}%</p>
            </div>
            <span className="text-2xl">⚡</span>
          </div>

          <div className="bg-[#4361ee] text-white p-4 rounded-xl flex items-center justify-between">
            <div>
              <p className="text-xs text-blue-200 font-semibold">Pack Temperature</p>
              <p className="text-3xl font-black mt-1">{liveData?.packTemp ?? vehicle.temp}°C</p>
            </div>
            <span className="text-2xl">🌡️</span>
          </div>

          <div className="bg-[#4361ee] text-white p-4 rounded-xl flex items-center justify-between">
            <div>
              <p className="text-xs text-blue-200 font-semibold">Odometer</p>
              <p className="text-2xl font-black mt-1">{vehicle.odometer} km</p>
            </div>
            <span className="text-2xl">📍</span>
          </div>

          <div className="bg-[#4361ee] text-white p-4 rounded-xl flex items-center justify-between">
            <div>
              <p className="text-xs text-blue-200 font-semibold">Cell Delta (Imbalance)</p>
              <p className="text-2xl font-black mt-1">{liveData?.cellImbalanceMv ?? 12} mV</p>
            </div>
            <span className="text-2xl">⚖️</span>
          </div>

          <div className="bg-[#4361ee] text-white p-4 rounded-xl flex items-center justify-between">
            <div>
              <p className="text-xs text-blue-200 font-semibold">Geofence Perimeter</p>
              <p className={`text-sm font-bold mt-1 ${isGeofenceAlert ? 'text-rose-300' : 'text-emerald-300'}`}>
                {isGeofenceAlert ? 'OUT OF BOUNDS' : 'AUTHORIZED'}
              </p>
            </div>
            <span className="text-2xl">{isGeofenceAlert ? '⚠️' : '✅'}</span>
          </div>
        </div>
      </div>

      {/* OTA Remote Controls Strip */}
      <RemoteOtaControls
        vehicleId={vehicle.id}
        currentIgnition={vehicle.ignition}
        onCommandExecuted={handleCommandAudit}
      />

      {/* Interactive Map with Geofencing & Real-Time / Playback Pin */}
      <div className="bg-white p-4 rounded-xl border border-slate-200 shadow-sm flex flex-col h-[400px]">
        <div className="flex justify-between items-center mb-2">
          <h4 className="text-xs font-bold text-slate-700 uppercase tracking-wide">
            Telemetry Mapping & Geofence Boundary
          </h4>
          <span className="text-xs text-slate-400 font-mono">
            Lat: {currentPoint.lat.toFixed(5)}, Lng: {currentPoint.lng.toFixed(5)}
          </span>
        </div>
        <div className="flex-1 overflow-hidden rounded-lg">
          <SpiroMap
            lat={currentPoint.lat}
            lng={currentPoint.lng}
            activeMarker={vehicle}
            routeHistory={VEHICLE_TRIP_HISTORY}
            currentPlaybackIndex={isPlaying ? playbackIndex : null}
            geofence={currentGeofence}
            geofenceAlert={isGeofenceAlert}
          />
        </div>
      </div>

      {/* Trip Replay Timeline Control Bar */}
      <TripPlaybackBar
        tripData={VEHICLE_TRIP_HISTORY}
        currentIndex={playbackIndex}
        onChangeIndex={setPlaybackIndex}
        isPlaying={isPlaying}
        setIsPlaying={setIsPlaying}
        speed={playbackSpeed}
        setSpeed={setPlaybackSpeed}
      />

      {/* Battery Degradation & Station Fleet Tabs */}
      <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">
        <div className="lg:col-span-5 bg-white p-4 rounded-xl border border-slate-200 shadow-sm">
          <h4 className="text-xs font-bold text-slate-700 uppercase tracking-wide mb-1">
            Battery Degradation & 80% EOL Curve
          </h4>
          <p className="text-[11px] text-slate-400 mb-3">Predicted retirement based on discharge cycle stresses</p>
          <DegradationChart data={MOCK_DEGRADATION_CURVE} />
        </div>

        <div className="lg:col-span-7">
          <SwapAndMaintenanceTabs />
        </div>
      </div>

      {/* Enterprise Security Audit Trail Log */}
      <AuditTrailDrawer newLogItem={latestAuditLog} />
    </div>
  );
}

```

### Verification and Edge Case Coverage

1. **Safety Alarm & Web Audio Chime:**

* If simulated packet temperature reaches or exceeds 60°C or cell voltage imbalance exceeds 195 mV, `triggerAlarmChime()` activates an 880Hz audio oscillation and displays the acknowledgment banner.

1. **Ray-Casting Geofence Validation:**

* Waypoint coordinate `[6.2300, 1.2650]` at timestamp `09:30` breaches the Lomé polygon perimeter ($lat > 6.22$), turning the map region red and changing the status card to `OUT OF BOUNDS`.

1. **Automated ESG Reporting:**

* The CSV exporter computes carbon offsets dynamically using the vehicle's odometer ($0.082 \text{ kg CO}_2\text{/km}$) and packages the waypoints into a downloadable file.

1. **Audit Trail Accountability:**

* Every OTA command (`LOCK`, `UNLOCK`, `REBOOT`) dispatches a record straight to the collapsible Audit Trail table with timestamp, operator ID, and completion status.
