Here is a production-grade, full-featured **EV Vehicle & Battery Fleet IoT Tracking Admin Portal** (**Spiro IoTHub**) built from scratch with **React 18/19**, modular custom hooks, an in-memory Axios mock API adapter layer, SVG charts/gauges, interactive maps, RBAC authentication/authorization, breadcrumbs, dense searchable/sortable tables, tabs, and drawer controls.

### 1. Mock API & Axios Interceptors (`src/api/`)

#### `src/api/axiosClient.js`

```javascript
import axios from 'axios';
import { handleMockRequest } from './mockAdapter';

const axiosClient = axios.create({
  baseURL: 'https://api.spiroiothub.internal/v1',
  timeout: 8000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// Request Interceptor: Attach JWT Token
axiosClient.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem('spiro_token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => Promise.reject(error)
);

// Response Interceptor: Route requests to Mock Adapter in development
axiosClient.interceptors.response.use(
  (response) => response,
  async (error) => {
    // Intercept network failures or mock routes
    if (error.config && (!error.response || error.response.status === 404)) {
      try {
        const mockData = await handleMockRequest(error.config);
        return { data: mockData, status: 200, statusText: 'OK', headers: {}, config: error.config };
      } catch (mockErr) {
        return Promise.reject(mockErr);
      }
    }
    if (error.response?.status === 401) {
      localStorage.removeItem('spiro_token');
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);

export default axiosClient;
```

#### `src/context/AuthContext.jsx`

```jsx
import React, { createContext, useContext, useState, useEffect } from 'react';
import axiosClient from '../api/axiosClient';

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(() => {
    const saved = localStorage.getItem('spiro_user');
    return saved ? JSON.parse(saved) : null;
  });
  const [token, setToken] = useState(() => localStorage.getItem('spiro_token'));

  const login = async (username, password) => {
    const res = await axiosClient.post('/auth/login', { username, password });
    const { token: jwt, user: userData } = res.data;
    setToken(jwt);
    setUser(userData);
    localStorage.setItem('spiro_token', jwt);
    localStorage.setItem('spiro_user', JSON.stringify(userData));
    return userData;
  };

  const logout = () => {
    setUser(null);
    setToken(null);
    localStorage.removeItem('spiro_token');
    localStorage.removeItem('spiro_user');
  };

  const hasPermission = (requiredRole) => {
    if (!user) return false;
    if (user.role === 'SUPER_ADMIN') return true;
    return user.role === requiredRole;
  };

  return (
    <AuthContext.Provider value={{ user, token, login, logout, hasPermission, isAuthenticated: !!token }}>
      {children}
    </AuthContext.Provider>
  );
}

export const useAuth = () => useContext(AuthContext);
```

#### `src/hooks/useApi.js`

```javascript
import { useState, useEffect, useCallback } from 'react';
import axiosClient from '../api/axiosClient';

export function useApi(endpoint, dependencies = []) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  const fetchData = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const res = await axiosClient.get(endpoint);
      setData(res.data);
    } catch (err) {
      setError(err.response?.data?.message || err.message || 'API Error');
    } finally {
      setLoading(false);
    }
  }, [endpoint]);

  useEffect(() => {
    fetchData();
  }, [...dependencies, fetchData]);

  return { data, loading, error, refetch: fetchData };
}
```

```javascript
import { useState, useEffect } from 'react';

export function useDebounce(value, delay = 300) {
  const [debouncedValue, setDebouncedValue] = useState(value);
  useEffect(() => {
    const timer = setTimeout(() => setDebouncedValue(value), delay);
    return () => clearTimeout(timer);
  }, [value, delay]);
  return debouncedValue;
}
```

```javascript
import { useState, useMemo } from 'react';

export function useTableSortFilter(initialData = [], config = {}) {
  const [search, setSearch] = useState('');
  const [sortKey, setSortKey] = useState(config.defaultSortKey || null);
  const [sortDirection, setSortDirection] = useState(config.defaultDirection || 'asc');
  const [filters, setFilters] = useState(config.initialFilters || {});

  const handleSort = (key) => {
    if (sortKey === key) {
      setSortDirection((prev) => (prev === 'asc' ? 'desc' : 'asc'));
    } else {
      setSortKey(key);
      setSortDirection('asc');
    }
  };

  const filteredSortedData = useMemo(() => {
    let result = [...initialData];

    // Filter by generic search query across designated keys
    if (search.trim()) {
      const query = search.toLowerCase();
      result = result.filter((item) => {
        const searchPool = config.searchKeys
          ? config.searchKeys.map((k) => item[k]).join(' ')
          : JSON.stringify(item);
        return searchPool.toLowerCase().includes(query);
      });
    }

    // Specific field filters (e.g. country, model, status)
    Object.entries(filters).forEach(([key, val]) => {
      if (val && val !== 'ALL') {
        result = result.filter((item) => String(item[key]).toLowerCase() === String(val).toLowerCase());
      }
    });

    // Sorting
    if (sortKey) {
      result.sort((a, b) => {
        let valA = a[sortKey];
        let valB = b[sortKey];
        if (typeof valA === 'string') valA = valA.toLowerCase();
        if (typeof valB === 'string') valB = valB.toLowerCase();
        if (valA < valB) return sortDirection === 'asc' ? -1 : 1;
        if (valA > valB) return sortDirection === 'asc' ? 1 : -1;
        return 0;
      });
    }

    return result;
  }, [initialData, search, sortKey, sortDirection, filters, config.searchKeys]);

  return {
    data: filteredSortedData,
    search,
    setSearch,
    sortKey,
    sortDirection,
    handleSort,
    filters,
    setFilter: (key, val) => setFilters((prev) => ({ ...prev, [key]: val })),
  };
}
```

```jsx
import React from 'react';
import { Routes, Route, Navigate } from 'react-router-dom';
import { useAuth } from './context/AuthContext';
import Layout from './components/layout/Layout';
import Login from './pages/Login';
import DashboardOverview from './pages/DashboardOverview';
import VehicleTracking from './pages/VehicleTracking';
import VehicleInfo from './pages/VehicleInfo';
import BatteryTracking from './pages/BatteryTracking';
import BatteryInfo from './pages/BatteryInfo';

function ProtectedRoute({ children }) {
  const { isAuthenticated } = useAuth();
  if (!isAuthenticated) return <Navigate to="/login" replace />;
  return children;
}

export default function App() {
  return (
    <Routes>
      <Route path="/login" element={<Login />} />

      <Route
        path="/"
        element={
          <ProtectedRoute>
            <Layout />
          </ProtectedRoute>
        }
      >
        <Route index element={<Navigate to="/overview" replace />} />
        <Route path="overview" element={<DashboardOverview />} />

        {/* Fallback */}

  );
}
```
