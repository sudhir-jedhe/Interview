```javascript
// Helper: Ray-casting algorithm to test if vehicle is inside polygon geofence
export function isPointInsidePolygon(point, polygon) {
  const [lat, lng] = point;
  let inside = false;
  for (let i = 0, j = polygon.length - 1; i < polygon.length; j = i++) {
    const xi = polygon[i][0], yi = polygon[i][1];
    const xj = polygon[j][0], yj = polygon[j][1];
    const intersect = yi > lng !== yj > lng && lat < ((xj - xi) * (lng - yi)) / (yj - yi) + xi;
    if (intersect) inside = !inside;
  }
  return inside;
}
```

```jsx
  geofence = null,
  geofenceAlert = false,
  showLegend = true,
  height = '100%',
}) {
  // Convert lat/lng to normalized SVG coordinate space (approx bounds: lat 6.08 to 6.25, lng 1.14 to 1.30)
  const mapCoordsToSvg = (pLat, pLng) => {
    const minLat = 6.08, maxLat = 6.26;
    const minLng = 1.14, maxLng = 1.30;
    const x = ((pLng - minLng) / (maxLng - minLng)) * 800;
    const y = 500 - ((pLat - minLat) / (maxLat - minLat)) * 500;
```

### 3. Trip History Playback Control Bar (`src/components/telemetry/TripPlaybackBar.jsx`)

```jsx
import React, { useEffect } from 'react';

export default function TripPlaybackBar({
  tripData = [],
  currentIndex,
  onChangeIndex,
  isPlaying,
  setIsPlaying,
  speed,
  setSpeed,
}) {
  useEffect(() => {
    let interval = null;
    if (isPlaying) {
      interval = setInterval(() => {
        onChangeIndex((prev) => {
          if (prev >= tripData.length - 1) {
            setIsPlaying(false);
            return prev;
          }
          return prev + 1;
        });
      }, 1000 / speed);
    }
    return () => clearInterval(interval);
  }, [isPlaying, speed, tripData.length, onChangeIndex, setIsPlaying]);

  const currentPoint = tripData[currentIndex] || tripData[0];

  return (
    <div className="bg-slate-900 text-white p-3.5 rounded-xl shadow-lg border border-slate-800 flex flex-col md:flex-row items-center justify-between gap-4">
      {/* Play/Pause & Speed Buttons */}
      <div className="flex items-center gap-3">
        <button
          onClick={() => {
            if (currentIndex >= tripData.length - 1) onChangeIndex(0);
            setIsPlaying(!isPlaying);
          }}
          className="w-9 h-9 flex items-center justify-center rounded-lg bg-blue-600 hover:bg-blue-500 transition text-sm font-bold shadow"
        >
          {isPlaying ? '⏸' : '▶'}
        </button>

        <div className="flex bg-slate-800 rounded-lg p-0.5 text-xs font-semibold">
          {[1, 2, 4].map((s) => (
            <button
              key={s}
              onClick={() => setSpeed(s)}
              className={`px-2 py-1 rounded transition ${
                speed === s ? 'bg-blue-600 text-white' : 'text-slate-400 hover:text-white'
              }`}
            >
              {s}x
            </button>
          ))}
        </div>
      </div>

      {/* Timeline Range Scrubber */}
      <div className="flex-1 w-full flex items-center gap-3">
        <span className="text-xs font-mono text-slate-400">
          {tripData[0]?.timestamp || '00:00'}
        </span>
        <input
          type="range"
          min="0"
          max={tripData.length - 1}
          value={currentIndex}
          onChange={(e) => onChangeIndex(Number(e.target.value))}
          className="w-full h-1.5 bg-slate-700 rounded-lg appearance-none cursor-pointer accent-blue-500"
        />
        <span className="text-xs font-mono text-slate-400">
          {tripData[tripData.length - 1]?.timestamp || '00:00'}
        </span>
      </div>

      {/* Instantaneous Sensor Metrics Readout */}

          <span className="text-[10px] text-slate-400 block">TIME</span>

          <span className="font-bold text-amber-400">{currentPoint?.soc}%</span>
        </div>
      </div>
    </div>
  );
}
```

```jsx
import React, { useState } from 'react';

export default function RemoteOtaControls({ vehicleId, currentIgnition, onCommandExecuted }) {
  const [isLocked, setIsLocked] = useState(currentIgnition === 'OFF');
  const [isPending, setIsPending] = useState(false);
  const [confirmAction, setConfirmAction] = useState(null); // 'LOCK' | 'UNLOCK' | 'REBOOT'

  const executeOtaCommand = (actionType) => {
    setIsPending(true);
    setConfirmAction(null);

    // Simulate OTA payload dispatch to MQTT / LoRaWAN gateway
    setTimeout(() => {
      setIsPending(false);
      if (actionType === 'LOCK') setIsLocked(true);
      if (actionType === 'UNLOCK') setIsLocked(false);
      if (onCommandExecuted) {
        onCommandExecuted(actionType);
      }
    }, 1200);
  };

  return (
    <div className="bg-white rounded-xl border border-slate-200 p-4 shadow-sm">
      <div className="flex items-center justify-between mb-3 border-b border-slate-100 pb-2">
        <h4 className="text-xs font-bold text-slate-700 uppercase tracking-wide">
          Remote OTA Telematics Command
        </h4>
        <span
          className={`text-[10px] font-bold px-2 py-0.5 rounded-full ${
            isLocked ? 'bg-rose-100 text-rose-700' : 'bg-emerald-100 text-emerald-700'
          }`}
        >
          {isLocked ? 'ENGINE IMMOBILIZED' : 'VEHICLE ACTIVE'}
        </span>
      </div>

      <div className="grid grid-cols-3 gap-2">
        {isLocked ? (
          <button
            disabled={isPending}
            onClick={() => setConfirmAction('UNLOCK')}
            className="py-2.5 px-3 rounded-lg bg-emerald-600 hover:bg-emerald-700 text-white font-bold text-xs flex items-center justify-center gap-1.5 transition disabled:opacity-50"
          >
            🔓 Remote Unlock
          </button>
        ) : (
          <button
            disabled={isPending}
            onClick={() => setConfirmAction('LOCK')}
            className="py-2.5 px-3 rounded-lg bg-rose-600 hover:bg-rose-700 text-white font-bold text-xs flex items-center justify-center gap-1.5 transition disabled:opacity-50"
          >
            🔒 Immobilize Engine
          </button>
        )}

        <button
          disabled={isPending}
          onClick={() => setConfirmAction('REBOOT')}
          className="py-2.5 px-3 rounded-lg bg-slate-100 hover:bg-slate-200 text-slate-700 font-bold text-xs flex items-center justify-center gap-1.5 transition disabled:opacity-50"
        >
          🔄 BMS Reboot
        </button>

        <button
          disabled={isPending}
          onClick={() => alert(`Honk/Blink command sent to ${vehicleId}`)}

          📢 Ping Horn
        </button>
      </div>

      {isPending && (
        <div className="mt-2 text-center text-xs text-blue-600 font-medium animate-pulse">
          Sending authenticated OTA command packet to Gateway...
        </div>
      )}

      {/* Confirmation Modal */}
      {confirmAction && (
        <div className="fixed inset-0 bg-slate-900/40 backdrop-blur-xs flex items-center justify-center z-50 p-4">
          <div className="bg-white rounded-xl max-w-sm w-full p-5 shadow-2xl border border-slate-200 text-left">
            <h3 className="font-bold text-slate-800 text-sm">
              Confirm {confirmAction} Command
            </h3>
            <p className="text-xs text-slate-500 mt-2">
              Are you sure you want to broadcast an immediate <b>{confirmAction}</b> instruction to vehicle{' '}
              <span className="font-mono text-blue-600">{vehicleId}</span>?
            </p>
            <div className="flex justify-end gap-2 mt-5">
              <button
                onClick={() => setConfirmAction(null)}
                className="px-3 py-1.5 rounded-lg text-xs font-semibold text-slate-600 hover:bg-slate-100"

                Cancel

              </button>
            </div>
          </div>

      )}
```
